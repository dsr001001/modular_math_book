\chapter{Groups, group actions and matrix groups over finite fields}\label{ch:groups}
\chaptermark{Groups and matrix groups over finite fields}

The paper uses group theory in two ways. Most of the time it is the background language: Galois groups
are profinite groups, inertia groups are stabilisers, Tamagawa numbers and component groups are finite
abelian groups whose $\ell$-parts are taken, and the tame Galois group of $\Q_p$ is a semidirect product.
At a few precise places, however, the paper's arguments are statements about specific finite groups: the
proof of Proposition prop:ns is a classification of certain subgroups of the group $\GL_2(\F_3)$ of order
$48$, of their subgroups of determinant one and of their orbits on the four lines of $\F_3^2$; Lemma lem:IV
and Remark rem:types are statements about unipotent elements of $\GL_2(\F_3)$; and Lemma lem:h0 uses that
$\F_\ell^\times$ is cyclic of order $\ell-1$. This chapter develops the elementary theory from the
definitions and then treats $\GL_2(\F_3)$ completely, so that every group-theoretic sentence of the paper
can be checked line by line. It relies on no earlier chapter, only on linear algebra over a field and on
the first notions of group theory as in Riley-Hobson-Bence.

\section*{Core results at a glance}

Throughout, $F$ is a field, $\F_q$ a finite field with $q$ elements ($q$ odd when Cartan subgroups appear;
the reader may take $q=p$ prime), and the standard models of \Cref{conv:groups:models} are used for the
subgroups of $\GL_2(\F_3)$.

\begin{enumerate}[label=\textbf{\thechapter.\arabic*}]
\item \textbf{Conventions.} Matrices act on column vectors on the left; $\PP^1(F)$ is the set of lines $[v]=Fv$ of $F^2$,
with $\infty=[1:0]=[e_1]$ and $x=[x:1]$, so that $\sm abcd$ acts by $x\mapsto(ax+b)/(cx+d)$; $D_n$ has order $2n$; $q$ is a
field size in this chapter and a bad prime in the paper; ``the image is $X$'' means ``conjugate to the standard model
$X$''. \Cref{conv:groups:conventions}.
\item \textbf{Groups} (definition: group, abelian group, subgroup, the subgroup $\langle S\rangle$ generated by $S$, the order $\#G$ of a group and the order
of an element, cyclic group). \Cref{def:groups:group}.
\item \textbf{Lagrange.} For $H\le G$ finite, the left cosets of $H$ partition $G$ into $[G:H]$ sets of $\#H$ elements, so
$\#G=[G:H]\,\#H$; the order of every element divides $\#G$. \emph{(proved here)} \Cref{prop:groups:lagrange}.
\item \textbf{Cyclic groups.} Let $C=\langle t\rangle$ have order $n$. (i) For each $d\mid n$, $C$ has exactly one subgroup of order $d$,
$C[d]=\{x:x^d=1\}=\langle t^{n/d}\rangle$, and every subgroup is of this form. (ii) $x\mapsto x^m$ has kernel $C[\gcd(m,n)]$ and
image $C^m=C^{\gcd(m,n)}$ of index $\gcd(m,n)$; so $[C:C^{\ell^k}]=\ell^{\min(k,v_\ell(n))}$ for a prime $\ell$. (iii) The generators
are the $\varphi(n)$ elements of order $n$, and $\Aut(C)\simeq(\Z/n)^\times$ via $t\mapsto t^a$. (iv) A surjection $\chi\colon G\to C$
satisfies $\chi(g)^2=1$ for all $g$ iff $n\mid 2$. \emph{(proved here)} \Cref{thm:groups:cyclic}.
\item \textbf{Normal subgroups and quotients} (definition: normal subgroup, quotient, kernel, centre $Z(G)$, centraliser $C_G(S)$,
normaliser $N_G(S)$, commutator subgroup $[G,G]$). \Cref{def:groups:normal}.
\item \textbf{Isomorphism theorems.} First, $G/\ker f\simeq f(G)$. Second, $HN/N\simeq H/(H\cap N)$ for $H\le G$ and $N\trianglelefteq G$. Third,
$(G/N)/(M/N)\simeq G/M$ for $N\le M$ normal. Correspondence: subgroups of $G/N$ correspond to subgroups of $G$ containing $N$, normal to normal. \emph{(proved here)}
\Cref{thm:groups:isothms}.
\item \textbf{Index two; image of a kernel.} A subgroup of index $2$ is normal. If $\rho\colon G\to\Gamma$ and $\delta\colon\Gamma\to A$ are
homomorphisms, then $\rho(\ker(\delta\circ\rho))=\rho(G)\cap\ker\delta$; in particular the image of $\ker(\det\circ\rho)$ is the
determinant-one subgroup of $\rho(G)$. \emph{(proved here)} \Cref{lem:groups:index2}.
\item \textbf{Conjugacy classes.} The class of $g$ has $[G:C_G(g)]$ elements; $H$ has $[G:N_G(H)]$ conjugates; a subgroup is normal
iff it is a union of classes. \emph{(proved here)} \Cref{prop:groups:conjclasses}.
\item \textbf{$\ell$-primary parts.} For a finite abelian group $A$ and a prime $\ell$, $A[\ell^\infty]=\{a:\ell^ka=0\text{ for some }k\}$ has order
$\ell^{v_\ell(\#A)}$, is the unique Sylow $\ell$-subgroup, and $A=\bigoplus_{\ell\mid\#A}A[\ell^\infty]$; Cauchy's theorem holds for $A$.
\emph{(proved here)} \Cref{prop:groups:lprimary}.
\item \textbf{Direct products.} If $H,K\trianglelefteq G$, $H\cap K=1$, $HK=G$, then $H\times K\simeq G$ by $(h,k)\mapsto hk$. \emph{(proved here)}
\Cref{prop:groups:directproduct}.
\item \textbf{Chinese remainder theorem for groups.} $C_m\times C_n\simeq C_{mn}$ iff $\gcd(m,n)=1$; $\Z/n\simeq\prod_{p\mid n}\Z/p^{v_p(n)}$.
\emph{(proved here)} \Cref{prop:groups:crtgroups}.
\item \textbf{Semidirect products} (definition: $N\rtimes_\theta H$, with $(n,h)(n',h')=(n\,\theta(h)(n'),hh')$ and $hnh^{-1}=\theta(h)(n)$).
\Cref{def:groups:semidirect}. \textbf{Recognition:} if $N\trianglelefteq G$, $H\le G$, $N\cap H=1$, $NH=G$, then $G\simeq N\rtimes_\theta H$ with $\theta$
conjugation; a short exact sequence $1\to N\to G\to Q\to1$ splits iff $G\simeq N\rtimes Q$. \emph{(proved here)}
\Cref{prop:groups:semidirect}.
\item \textbf{Normal forms.} If $G=\{a^mb^e:0\le m<n,\ e\in\{0,1\}\}$ ($2n$ distinct elements) with $a^n=1$, $bab^{-1}=a^k$, $b^2=a^c$, the multiplication of $G$ is
determined by $(n,k,c)$; two such groups with the same $(n,k,c)$ are isomorphic via $a^mb^e\mapsto a'^mb'^e$. \emph{(proved here)} \Cref{lem:groups:normalform}.
\item \textbf{Dihedral groups.} $D_n=\langle r,s\mid r^n=s^2=1,\ srs^{-1}=r^{-1}\rangle\simeq C_n\rtimes C_2$ has order $2n$; all elements outside
$\langle r\rangle$ have order $2$; a group of order $2n$ generated by $r$ of order $n$ and $s$ of order $2$ with $srs^{-1}=r^{-1}$ is $\simeq D_n$;
$D_3\simeq S_3$, $D_6\simeq C_2\times S_3$; $D_4$ has five involutions. \emph{(proved here)} \Cref{def:groups:dihedral},
\Cref{prop:groups:dihedral}.
\item \textbf{The quaternion group.} $Q_8=\{\pm1,\pm i,\pm j,\pm k\}$, $i^2=j^2=k^2=ijk=-1$: $-1$ is the only involution, $Z(Q_8)=[Q_8,Q_8]=\{\pm1\}$,
every subgroup is normal, $Q_8\not\simeq D_4$; a non-abelian group of order $8$ with a unique involution is $\simeq Q_8$. \emph{(proved
here)} \Cref{def:groups:quaternion}, \Cref{prop:groups:quaternion}.
\item \textbf{The semidihedral group.} $SD_{16}=\langle a,x\mid a^8=x^2=1,\ xax^{-1}=a^3\rangle$ has order $16$; $(a^ix)^2=a^{4i}$; it has $1,5,6,4$
elements of orders $1,2,4,8$; its index-$2$ subgroups are $\langle a\rangle\simeq C_8$, $\langle a^2,x\rangle\simeq D_4$, $\langle a^2,ax\rangle\simeq Q_8$; a group of
order $16$ generated by $a$ of order $8$ and $x$ of order $2$ with $xax^{-1}=a^3$ is $\simeq SD_{16}$. \emph{(proved here)}
\Cref{def:groups:semidihedral}, \Cref{prop:groups:semidihedral}.
\item \textbf{Groups of order 8.} Every group of order $8$ is isomorphic to exactly one of $C_8$, $C_4\times C_2$, $C_2^3$, $D_4$, $Q_8$.
\emph{(proved here)} \Cref{thm:groups:order8}.
\item \textbf{Group actions} (definition: action, orbit $Gx$, stabiliser $G_x$, fixed set $X^G$, transitive, faithful).
\Cref{def:groups:action}.
\item \textbf{Orbit-stabiliser.} $gG_x\mapsto gx$ is a bijection $G/G_x\to Gx$; $G_{gx}=gG_xg^{-1}$; $\#X=\sum_i[G:G_{x_i}]$ over orbit
representatives. \emph{(proved here)} \Cref{thm:groups:orbitstab}.
\item \textbf{Orbits of a subgroup.} If $G$ is transitive on $X$ and $H\le G$, then $H$ is transitive iff $G=HG_x$; if $H\trianglelefteq G$, the $H$-orbits are
permuted transitively by $G$ and have equal size; if $[G:H]=2$, either $H$ is transitive or $X$ is the union of two $H$-orbits of
size $\#X/2$, interchanged by $G\setminus H$. \emph{(proved here)} \Cref{prop:groups:suborbits}.
\item \textbf{Orbit counting.} The number of orbits of a finite group on a finite set is $\frac1{\#G}\sum_g\#X^g$. \emph{(proved here)}
\Cref{lem:groups:burnside}.
\item \textbf{Class equation and $p$-groups.} $\#G=\#Z(G)+\sum[G:C_G(g_i)]$; if a $p$-group acts on $X$, $\#X\equiv\#X^P\pmod p$; nontrivial
$p$-groups have nontrivial centre; groups of order $p^2$ are abelian; a group of order $p^n$ has normal subgroups of every order
$p^k$, and subgroups of index $p$ in a $p$-group are normal. \emph{(proved here)} \Cref{thm:groups:classeq}.
\item \textbf{Cauchy.} If a prime $p$ divides $\#G$, $G$ has an element of order $p$. \emph{(proved here)} \Cref{thm:groups:cauchy}.
\item \textbf{Sylow.} Let $\#G=p^am$, $p\nmid m$. Sylow $p$-subgroups (order $p^a$) exist; every $p$-subgroup lies in one; they are all
conjugate; their number $n_p=[G:N_G(P)]$ divides $m$ and is $\equiv1\pmod p$; $P\trianglelefteq G$ iff $n_p=1$. \emph{(proved here)}
\Cref{thm:groups:sylow}.
\item \textbf{Finite abelian groups.} $A\simeq\bigoplus_i\Z/\ell_i^{e_i}$ with the multiset $\{\ell_i^{e_i}\}$ unique; equivalently $A\simeq\bigoplus\Z/d_i$
with $1<d_1\mid\dots\mid d_r$ unique; the numbers $\#A[\ell^k]$ determine $A[\ell^\infty]$. \emph{(proved here)} \Cref{thm:groups:finiteabelian}.
\item \textbf{Cyclicity criteria.} A finite abelian group is cyclic iff $\#A[\ell]\le\ell$ for all primes $\ell$; every finite subgroup of
$K^\times$, $K$ a field, is cyclic; so $\F_q^\times$ and $\mu_n(K)$ are cyclic. \emph{(proved here)} \Cref{cor:groups:cyclic}.
\item \textbf{Counting torsion.} $\#A[n]=\#A/nA$; for $A\simeq\bigoplus_i\Z/\ell^{e_i}$, $\#A[\ell^k]=\ell^{\sum_i\min(k,e_i)}$. \emph{(proved here)}
\Cref{prop:groups:torsioncount}.
\item \textbf{Linear groups} (definition: $\GL_2(F)$, $\SL_2(F)$, $\PGL_2(F)=\GL_2(F)/F^\times I$, $\PSL_2(F)=\SL_2(F)/\{\pm I\}$).
\Cref{def:groups:gl2}. $Z(\GL_2(F))=F^\times I$, $Z(\SL_2(F))=\{\pm I\}$; $\GL_2(F)\simeq\SL_2(F)\rtimes F^\times$; $\SL_2(F)$ is generated by
elementary matrices; for $\operatorname{char}F\ne2$, $-I$ is the only involution of $\SL_2(F)$; $[\PGL_2(F):\PSL_2(F)]=[F^\times:(F^\times)^2]$.
\emph{(proved here)} \Cref{prop:groups:gl2}.
\item \textbf{Companion form.} Two non-scalar matrices in $\GL_2(F)$ are conjugate iff they have the same characteristic polynomial;
the centraliser of a non-scalar $g$ is $F[g]^\times$. \emph{(proved here)} \Cref{lem:groups:companion}.
\item \textbf{Orders.} $\#\GL_2(\F_q)=q(q-1)^2(q+1)$, $\#\SL_2(\F_q)=\#\PGL_2(\F_q)=q(q^2-1)$, $\#\PSL_2(\F_q)=q(q^2-1)/\gcd(2,q-1)$
($48,24,24,12$ for $q=3$). \emph{(proved here)} \Cref{prop:groups:orders}.
\item \textbf{The projective line.} $\GL_2(F)$ acts on $\PP^1(F)$ with kernel $F^\times I$; $\PGL_2(F)$ is sharply $3$-transitive; the stabiliser of
$\infty$ is the upper triangular group $\mathrm B(F)$; $\SL_2(F)$ is $2$-transitive; $\#\PP^1(\F_q)=q+1$. \emph{(proved here)}
\Cref{prop:groups:projline}.
\item \textbf{Borel subgroups} (definition: the stabiliser $\mathrm B_L$ of a line $L$; $\mathrm U$ the upper unitriangular group, $\mathrm T$ the
diagonal group). \Cref{def:groups:borel}. $\mathrm B=\mathrm U\rtimes\mathrm T$, $\#\mathrm B(\F_q)=q(q-1)^2$, Borel subgroups are conjugate and self-normalising,
and a subgroup stabilises a line iff it lies in a Borel subgroup; $\mathrm B(\F_3)\simeq C_2\times S_3\simeq D_6$. \emph{(proved here)}
\Cref{prop:groups:borel}.
\item \textbf{Unipotent elements} (definition: $(u-1)^2=0$). \Cref{def:groups:unipotent}. A unipotent $u\ne1$ is conjugate to $\sm1101$ and
fixes exactly one line, pointwise; in characteristic $p$ the unipotents $\ne1$ are exactly the elements of $p$-power order $\ne1$,
and have order $p$; $-u$ fixes no nonzero vector ($\operatorname{char}F\ne2$), stabilises one line and has order $2p$; the stabiliser of a
nonzero vector $v$ in $\SL_2(F)$ is the group $\mathrm U_{[v]}\simeq(F,+)$; in $\GL_2(\F_3)$ the $8$ elements of order $3$ are the
nontrivial unipotents, form one class, lie in the four Sylow $3$-subgroups $\mathrm U_L$, and generate $\SL_2(\F_3)$; the $8$
elements of order $6$ are the $-u$. \emph{(proved here)} \Cref{prop:groups:unipotent}.
\item \textbf{Commutators in a Borel subgroup.} For $H\le\mathrm B(K)$, $[H,H]\le\mathrm U(K)$. If $\operatorname{char}K=p$: a subgroup of $\mathrm B(K)$ of order prime to $p$
is abelian, and a finite subgroup of $\GL_2(K)$ whose commutator subgroup is not a $p$-group stabilises no line of $K^2$.
\emph{(proved here)} \Cref{lem:groups:borelcommutator}.
\item \textbf{$\PGL_2$ in small characteristic.} $\PGL_2(\F_3)\simeq S_4$ and $\PSL_2(\F_3)\simeq A_4$ through the four lines; for $g\in\GL_2(\F_3)$,
$\det g$ is the sign of the permutation of the lines. \emph{(proved here)} $\PGL_2(\F_5)\simeq S_5$, $\PSL_2(\F_5)\simeq A_5$. \emph{(proof
sketched here)} $\PSL_2(\F_q)$ is simple for $q\ge4$, and $\PSL_2(\F_7)\simeq\GL_3(\F_2)$. \emph{(stated; reference)}
\Cref{thm:groups:smallpgl}.
\item \textbf{Cartan subgroups} (definition, $q$ odd, $\varepsilon$ a non-square: $C_s=\{\mathrm{diag}(a,d)\}$; $C_{ns}=\{\sm a{\varepsilon b}ba:(a,b)\ne(0,0)\}\simeq\F_q[\sqrt\varepsilon]^\times$).
\Cref{def:groups:cartan}. $N(C_s)=C_s\cup wC_s$ is the stabiliser of $\{[e_1],[e_2]\}$; $N(C_{ns})=C_{ns}\cup C_{ns}\sigma$ with $\sigma=\mathrm{diag}(1,-1)$ and
$\sigma c\sigma^{-1}=c^q$; $C_{ns}$ is cyclic of order $q^2-1$; all split, respectively all nonsplit, Cartan subgroups are conjugate.
\emph{(proved here)} \Cref{prop:groups:cartan}.
\item \textbf{Cartan facts.} $\#C_s=(q-1)^2$, $\#N_s=2(q-1)^2$, $\#C_{ns}=q^2-1$, $\#N_{ns}=2(q^2-1)$; elements of $N_s\setminus C_s$ and of
$N_{ns}\setminus C_{ns}$ have trace $0$, with $(c\sigma)^2=\det(c)I$; $\det(C_{ns})=\F_q^\times$; the determinant-one subgroups have orders
$q-1$, $2(q-1)$, $q+1$, $2(q+1)$; orbits on $\PP^1(\F_q)$: $C_s$: $1,1,q-1$; $N_s$: $2,q-1$; $\mathrm B$: $1,q$; $C_{ns}$ and $N_{ns}$ transitive.
\emph{(proved here)} \Cref{prop:groups:cartanfacts}.
\item \textbf{Dickson.} For $\ell$ odd and $G\le\GL_2(\F_\ell)$ with image $\bar G$ in $\PGL_2(\F_\ell)$: if $\ell\mid\#G$, then $G\supseteq\SL_2(\F_\ell)$ or $G$ lies in a Borel
subgroup; if $\ell\nmid\#G$, then $\bar G$ is cyclic and $G$ lies in a Cartan subgroup, or $\bar G$ is dihedral and $G$ lies in the normaliser
of a Cartan subgroup but not in the Cartan subgroup, or $\bar G\simeq A_4,S_4,A_5$. \emph{(stated; reference; part (a) for $\ell=3$ is
proved in \Cref{prop:groups:gl23subgroups}, and the case needed by the paper in \Cref{thm:groups:threeimages})} \Cref{thm:groups:dickson}.
\item \textbf{Conjugacy classes of $\GL_2(\F_3)$.} Order $48$, centre $\{\pm I\}$, commutator subgroup $\SL_2(\F_3)$, eight classes, of sizes
$1,1,12,8,6,8,6,6$ with representatives $I,-I,\sigma,u,J,-u,\sm1{-1}11,\sm0111$ of orders $1,2,2,3,4,6,8,8$. \emph{(proved here)}
\Cref{prop:groups:gl23classes}.
\item \textbf{Standard models.} $J=\sm0{-1}10$, $\sigma=\mathrm{diag}(1,-1)$, $w=\sm0110$, $u=\sm1101$; $C_s(\F_3)=\{\mathrm{diag}(\pm1,\pm1)\}$,
$N_s(\F_3)=\langle J,\sigma\rangle$, $C_{ns}(\F_3)=\{\sm a{-b}ba\}$, $N_{ns}(\F_3)=C_{ns}(\F_3)\cup C_{ns}(\F_3)\sigma$, $Q_8=\{\pm I,\pm J,\pm\sm111{-1},\pm\sm1{-1}{-1}{-1}\}$,
$C_4=\langle J\rangle$. \Cref{conv:groups:models}.
\item \textbf{Subgroups of $\GL_2(\F_3)$.} There are $55$ subgroups in $16$ conjugacy classes (table); $9$ classes have surjective
determinant, and all of these except $C_{ns}(\F_3)$ contain an element of determinant $-1$ and trace $0$; a subgroup of order divisible
by $3$ contains $\SL_2(\F_3)$ or stabilises a line. \emph{(proved here for the rows used; the table checked by computer)}
\Cref{prop:groups:gl23subgroups}.
\item \textbf{$\SL_2(\F_3)$.} Order $24$; $-I$ its only involution; its elements of order dividing $4$ form its unique Sylow $2$-subgroup
$Q_8$, normal in $\GL_2(\F_3)$; $\SL_2(\F_3)=Q_8\rtimes\langle u\rangle$ with $u$ permuting $i,j,k$ cyclically; $[\SL_2(\F_3),\SL_2(\F_3)]=Q_8$; no
subgroup of order $12$; $\GL_2(\F_3)/Q_8\simeq S_3$; $Q_8$ is transitive on the four lines. \emph{(proved here)} \Cref{prop:groups:sl23}.
\item \textbf{$N_s(\F_3)$.} $N_s(\F_3)=\langle J,\sigma\rangle\simeq D_4$; its elements of determinant $-1$ are $\pm\sigma,\pm w$, of trace $0$; it has $3$
conjugates; $N_s(\F_3)\cap\SL_2(\F_3)=\langle J\rangle\simeq C_4$; its orbits on the four lines are $\{\infty,0\}$ and $\{1,-1\}$, and $\langle J\rangle$ has
the same orbits; it stabilises no line of $\Fbar_3^2$, while $\langle J\rangle$ stabilises the two lines of $\F_9^2$ spanned by $(\pm i,1)$;
$N_s(\F_3)$ has index $2$ in $N_{ns}(\F_3)$. \emph{(proved here)} \Cref{prop:groups:ns3}.
\item \textbf{$N_{ns}(\F_3)$.} $C_{ns}(\F_3)=\langle a\rangle\simeq C_8$ with $a=\sm1{-1}11$; $N_{ns}(\F_3)=\langle a,\sigma\rangle\simeq SD_{16}$, $\sigma a\sigma^{-1}=a^3$; it is a Sylow
$2$-subgroup of $\GL_2(\F_3)$; $N_{ns}(\F_3)\cap\SL_2(\F_3)=Q_8$; its elements of determinant $-1$ and trace $0$ are $\pm\sigma,\pm w$, and
$C_{ns}(\F_3)$ has none; $C_{ns}(\F_3)$, $N_{ns}(\F_3)$ and $Q_8$ are transitive on the four lines. \emph{(proved here)} \Cref{prop:groups:nns3}.
\item \textbf{The three images.} If $H\le\GL_2(\F_3)$ has $\det(H)=\F_3^\times$, stabilises no line of $\F_3^2$, and contains an element of
determinant $-1$ and trace $0$, then $H$ is conjugate to exactly one of $\GL_2(\F_3)$, $N_{ns}(\F_3)$, $N_s(\F_3)$; the determinant-one
subgroups are $\SL_2(\F_3)$, $Q_8$, $\langle J\rangle$; the first two stabilise no line of $\Fbar_3^2$, the third is abelian. \emph{(proved here)}
\Cref{thm:groups:threeimages}.
\item \textbf{Orbit dichotomy.} For such $H$: $H$ is conjugate to $N_s(\F_3)$ iff it has two orbits of size $2$ on the four lines, iff it
is not transitive, iff $H\cap\SL_2(\F_3)$ is abelian, iff $H\cap\SL_2(\F_3)$ stabilises a line of $\Fbar_3^2$. \emph{(proved here)}
\Cref{cor:groups:orbitdichotomy}.
\item \textbf{Determinant $-1$, trace $0$.} For $q$ odd, $g\in\GL_2(\F_q)$ has $\det g=-1$ and $\tr g=0$ iff it is conjugate to $\sigma$; these $q(q+1)$
elements have order $2$ and fix exactly two lines; in $\GL_2(\F_3)$, $\mathrm B$, $N_s$, $N_{ns}$, $C_s$, $C_{ns}$, $\SL_2$ there are $12,6,4,4,2,0,0$ of them.
\emph{(proved here)} \Cref{prop:groups:cc}.
\item \textbf{Images in $S_4$.} Under $\PGL_2(\F_3)\simeq S_4$: $\GL_2,\SL_2,N_{ns},Q_8,N_s,C_{ns},C_s,\langle J\rangle,\mathrm B,\mathrm U$ map onto $S_4$, $A_4$, $D_4$,
the normal $V_4$, a non-normal $V_4$, $C_4$, $\langle$transposition$\rangle$, $\langle$double transposition$\rangle$, $S_3$, $A_3$. \emph{(proved here)}
\Cref{prop:groups:imagesS4}.
\item \textbf{Inverse limits} (definition: $\varprojlim G_n=\{(g_n):\pi_n(g_{n+1})=g_n\}$, profinite group). \Cref{def:groups:inverselimit}. The inverse
limit is a closed subgroup of $\prod G_n$, compact and Hausdorff, with the universal property. \emph{(proved here, except Tychonoff's
theorem: stated; reference)} \Cref{prop:groups:inverselimit}.
\item \textbf{Profinite groups.} With $U_n=\ker(G\to G_n)$: the $U_n$ are a basis of neighbourhoods of $1$; open subgroups are the closed
subgroups of finite index; continuous maps to finite groups factor through some $G/U_n$; closed subgroups and quotients by closed
normal subgroups are profinite; continuous images in Hausdorff groups are closed; a continuous homomorphism from a pro-$p$ group
to a finite group of order prime to $p$ is trivial. \emph{(proved here)} Profinite iff compact, Hausdorff and totally disconnected.
\emph{(stated; reference)} \Cref{prop:groups:profinite}.
\item \textbf{$\GL_2(\Z_\ell)$.} $\GL_2(\Z_\ell)=\varprojlim\GL_2(\Z/\ell^n)$; the reduction maps are surjective, with kernels isomorphic to $(M_2(\F_\ell),+)$, so
$\#\GL_2(\Z/\ell^n)=\ell^{4(n-1)}(\ell^2-1)(\ell^2-\ell)$; the kernel $\Gamma(\ell)=\ker(\GL_2(\Z_\ell)\to\GL_2(\F_\ell))$ is pro-$\ell$, and torsion-free if $\ell$ is odd, so a
finite subgroup of $\GL_2(\Z_\ell)$ maps isomorphically into $\GL_2(\F_\ell)$ for $\ell$ odd. \emph{(proved here)} \Cref{prop:groups:gl2zl}.
\end{enumerate}

\section{Groups, subgroups, cosets, quotients and cyclic groups}\label{sec:groups:basics}

This section fixes the conventions of the chapter and proves the basic facts about cosets, quotients
and cyclic groups. In the paper they appear in three places: the image of the absolute Galois group of
$\Q(\sqrt{-3})$ is a determinant-one subgroup (Proposition prop:ns, through \Cref{lem:groups:index2}); for an odd prime $\ell$, a character
of $G_\Q$ onto the cyclic group $\F_\ell^\times$ with square $1$ forces $\ell=3$ (Lemma lem:h0, through
\Cref{thm:groups:cyclic}); and the ``$\ell$-part'' of a finite abelian group (Definition def:terms, Lemma lem:tamfactor,
through \Cref{prop:groups:lprimary}).

\begin{convention}\label{conv:groups:conventions}
The following conventions hold in this chapter and are inherited by the book.
\begin{enumerate}[label=(C\arabic*)]
\item Matrices act on column vectors on the left. For $v\in F^2\setminus\{0\}$ the line through $v$ is $[v]=Fv$, and $\PP^1(F)$ is the set of
lines of $F^2$, with coordinates $[x:y]$. We write $\infty=[1:0]=[e_1]$ and identify $[x:1]$ with $x\in F$. Then
$g=\sm abcd$ acts by the M\"obius formula $g\cdot x=(ax+b)/(cx+d)$, because $g\binom x1=\binom{ax+b}{cx+d}$.
\item Conjugation is $g\cdot h=ghg^{-1}$. In a semidirect product $N\rtimes H$ the group $H$ acts on $N$, and $hnh^{-1}=\theta(h)(n)$.
\item $C_n$, or $\Z/n$ when written additively, is the cyclic group of order $n$; $D_n$ is the dihedral group of order $2n$ (so $D_4$
has order $8$; Dummit and Foote write $D_8$); $V_4=C_2\times C_2$; $Q_8$ is the quaternion group; $SD_{16}$ is the semidihedral group
of order $16$; $S_n$ and $A_n$ are the symmetric and alternating groups.
\item $\F_p=\Z/p\Z$ is a field: an integer prime to $p$ is invertible modulo $p$ by Bezout's identity. $\F_q$ is a finite field with
$q$ elements, of characteristic $p$ with $q$ a power of $p$ (\Cref{ch:rings} constructs it); the reader may take $q=p$
throughout. Whenever Cartan subgroups are discussed $q$ is odd. In this chapter $q$ is a field size; in the paper $q$ is a bad prime
of $E$ (convention B-C1). The two meet only in $\GL_2(\F_3)$, where the field size is $\ell=3$.
\item ``The image is $N_s(\F_3)$'' always means ``the image is conjugate in $\GL_2(\F_3)$ to $N_s(\F_3)$''; the standard models of
\Cref{conv:groups:models} are fixed once and for all, and similarly for $N_{ns}(\F_3)$, $C_s$, $C_{ns}$, $\mathrm B$, $\mathrm U$.
\end{enumerate}
\end{convention}

\begin{definition}\label{def:groups:group}
A \emph{group} is a set $G$ with an associative composition $G\times G\to G$, $(g,h)\mapsto gh$, an identity element $1$ and inverses.
It is \emph{abelian} if $gh=hg$ for all $g,h$; abelian groups are often written additively, with $0$ and $-a$. A \emph{subgroup} $H\le G$
is a nonempty subset closed under products and inverses. The subgroup $\langle S\rangle$ generated by a subset $S$ is the intersection of
all subgroups containing $S$, that is, the set of finite products of elements of $S$ and their inverses. The \emph{order} of $G$ is
its cardinality $\#G$; the order of $g\in G$ is the least $n\ge1$ with $g^n=1$ (or $\infty$), which is $\#\langle g\rangle$. A group is \emph{cyclic}
if $G=\langle g\rangle$ for some $g$. Examples: $(\Z,+)$; $\Z/n$; the symmetric group $S_n$; the group $\GL_2(F)$ of invertible $2\times2$
matrices over a field $F$; the multiplicative group $F^\times$.
\end{definition}

\begin{proposition}[Lagrange]\label{prop:groups:lagrange}
Let $H\le G$ with $G$ finite. The left cosets $gH$ partition $G$, each has $\#H$ elements, and $\#G=[G:H]\cdot\#H$, where $[G:H]$ is the
number of cosets. Consequently the order of every element of $G$ divides $\#G$, and $g^{\#G}=1$ for every $g\in G$.
\end{proposition}

\begin{proof}
Every $g$ lies in $gH$. If $x\in gH\cap g'H$, say $x=gh=g'h'$, then $g'=gh h'^{-1}\in gH$, so $g'H\subseteq gH$, and symmetrically; so two cosets
are equal or disjoint. The map $H\to gH$, $h\mapsto gh$, is bijective (its inverse is multiplication by $g^{-1}$). Counting gives
$\#G=[G:H]\,\#H$. Applied to $H=\langle g\rangle$, the order of $g$ divides $\#G$, and then $g^{\#G}=1$.
\end{proof}

\begin{theorem}\label{thm:groups:cyclic}
Let $C=\langle t\rangle$ be cyclic of order $n$.
\begin{enumerate}[label=(\roman*)]
\item Every subgroup of $C$ is cyclic. For each $d\mid n$ there is exactly one subgroup of order $d$, namely
$C[d]=\{x\in C:x^d=1\}=\langle t^{n/d}\rangle=C^{n/d}$, the set of $(n/d)$-th powers.
\item For $m\ge1$ the map $x\mapsto x^m$ is an endomorphism of $C$ with kernel $C[\gcd(m,n)]$, of order $\gcd(m,n)$, and image
$C^m=C^{\gcd(m,n)}$, of index $\gcd(m,n)$. In particular, for a prime $\ell$ and $k\ge0$, $[C:C^{\ell^k}]=\ell^{\min(k,v_\ell(n))}$; and for
$k\le v_\ell(n)$, an element $x\in C$ is an $\ell^k$-th power iff its image in $C/C^{\ell^{v_\ell(n)}}\simeq\Z/\ell^{v_\ell(n)}$ is divisible by $\ell^k$.
\item The generators of $C$ are the $\varphi(n)$ elements $t^a$ with $\gcd(a,n)=1$, which are its elements of order $n$. The map
$(\Z/n)^\times\to\Aut(C)$, $a\mapsto(t\mapsto t^a)$, is an isomorphism. $\Aut(\Z)=\{\pm1\}$.
\item Let $\chi\colon G\to C$ be a surjective homomorphism from any group. Then $\chi(g)^2=1$ for all $g\in G$ if and only if $n\mid2$.
\end{enumerate}
\end{theorem}

\begin{proof}
The map $\Z\to C$, $a\mapsto t^a$, is a surjective homomorphism with kernel $n\Z$ (if $t^a=1$, write $a=bn+r$ with $0\le r<n$; then
$t^r=1$ and $r=0$ by minimality of $n$). So $t^a=t^b$ iff $a\equiv b\pmod n$, and the order of $t^a$ is $n/\gcd(a,n)$: indeed $(t^a)^k=1$ iff
$n\mid ak$ iff $\frac n{\gcd(a,n)}\mid k$.

(i) Let $H\le C$, $H\ne1$, and let $a$ be the least positive integer with $t^a\in H$. For $t^b\in H$ write $b=ca+r$, $0\le r<a$; then
$t^r=t^b(t^a)^{-c}\in H$, so $r=0$. Hence $H=\langle t^a\rangle$, and $a\mid n$ (take $b=n$). So $H=\langle t^a\rangle$ has order $n/a$: the subgroups are the
$\langle t^{n/d}\rangle$ for $d\mid n$, one for each order $d$. The subgroup $\langle t^{n/d}\rangle$ consists of $(n/d)$-th powers and has order $d$, so its elements
satisfy $x^d=1$; conversely $x=t^b$ with $x^d=1$ means $n\mid bd$, that is $\frac nd\mid b$. So $C[d]=\langle t^{n/d}\rangle=C^{n/d}$.

(ii) The map is a homomorphism because $C$ is abelian. Let $g=\gcd(m,n)$. Its kernel is $C[m]$; and $x^m=1$ iff $x^g=1$ (write
$g=am+bn$, so $x^g=(x^m)^a(x^n)^b$), so the kernel is $C[g]$, of order $g$ by (i). By the first isomorphism theorem
(\Cref{thm:groups:isothms}, whose proof does not use this one) the image has order $n/g$; it is contained in $C^g$ (as $g\mid m$), which
has order $n/g$ by (i) applied to $d=n/g$; so $C^m=C^g$, of index $g$. For $m=\ell^k$, $g=\ell^{\min(k,v_\ell(n))}$. For the last statement put
$e=v_\ell(n)$ and $k\le e$: $C^{\ell^k}$ contains $C^{\ell^e}$, and under $C\to C/C^{\ell^e}$, a cyclic group of order $\ell^e$ generated by the image of
$t$, the subgroup $C^{\ell^k}$ maps onto the $\ell^k$-th powers, the subgroup of index $\ell^k$, which is the set of multiples of $\ell^k$
when $C/C^{\ell^e}$ is identified with $\Z/\ell^e$ through $t\mapsto1$. Since $C^{\ell^k}$ contains the kernel $C^{\ell^e}$, $x\in C^{\ell^k}$ iff its image is
divisible by $\ell^k$.

(iii) $t^a$ generates iff its order $n/\gcd(a,n)$ is $n$. An automorphism $\alpha$ is determined by $\alpha(t)$, which must be a generator
$t^a$; conversely $x\mapsto x^a$ with $\gcd(a,n)=1$ is a bijective endomorphism by (ii). Composition corresponds to multiplication of
exponents. For $\Z$, an automorphism sends the generator $1$ to a generator, $\pm1$.

(iv) If $n\mid2$, every element of $C$ satisfies $x^2=1$. Conversely, if $\chi(g)^2=1$ for all $g$, then, $\chi$ being surjective, every
$x\in C$ satisfies $x^2=1$; in particular $t^2=1$ and $n\mid2$.
\end{proof}

\begin{definition}\label{def:groups:normal}
A subgroup $N\le G$ is \emph{normal}, $N\trianglelefteq G$, if $gNg^{-1}=N$ for all $g\in G$. Then the cosets $gN$ form a group $G/N$, the
\emph{quotient}, with $(gN)(g'N)=gg'N$, and $G\to G/N$ is the \emph{canonical projection}. A \emph{homomorphism} $f\colon G\to H$ satisfies
$f(gg')=f(g)f(g')$; its \emph{kernel} $\ker f=f^{-1}(1)$ is normal and its image $f(G)$ is a subgroup. A group is \emph{simple} if it is
nontrivial and has no normal subgroups other than $1$ and itself. The \emph{centre} is $Z(G)=\{z:zg=gz\ \forall g\}$; for $S\subseteq G$, the
\emph{centraliser} is $C_G(S)=\{g:gs=sg\ \forall s\in S\}$ and the \emph{normaliser} is $N_G(S)=\{g:gSg^{-1}=S\}$. The \emph{conjugacy class} of $g$ is
$\{hgh^{-1}:h\in G\}$. The \emph{commutator subgroup} $[G,G]$ is generated by the commutators $[g,h]=ghg^{-1}h^{-1}$; it is normal, $G/[G,G]$
is abelian (the \emph{abelianisation}), and $[G,G]\le N$ for a normal $N$ iff $G/N$ is abelian.
\end{definition}

The multiplication on $G/N$ is well defined because $gNg'N=gg'(g'^{-1}Ng')N=gg'N$. The last statement of the definition holds
because $G/N$ is abelian iff $ghg^{-1}h^{-1}\in N$ for all $g,h$.

\begin{theorem}[Isomorphism theorems]\label{thm:groups:isothms}
\begin{enumerate}[label=(\roman*)]
\item A homomorphism $f\colon G\to H$ induces an isomorphism $G/\ker f\simeq f(G)$, $g\ker f\mapsto f(g)$.
\item If $H\le G$ and $N\trianglelefteq G$, then $HN$ is a subgroup, $H\cap N\trianglelefteq H$, and $HN/N\simeq H/(H\cap N)$.
\item If $N\le M$ are normal subgroups of $G$, then $M/N\trianglelefteq G/N$ and $(G/N)/(M/N)\simeq G/M$.
\item (Correspondence) For $N\trianglelefteq G$, $K\mapsto K/N$ is an inclusion-preserving bijection from the subgroups of $G$ containing $N$ to
the subgroups of $G/N$, under which normal subgroups correspond to normal subgroups.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) $f(g)=f(g')$ iff $g^{-1}g'\in\ker f$ iff $g\ker f=g'\ker f$; so the map is well defined and injective; it is a surjective
homomorphism onto $f(G)$. (ii) $hnh'n'=hh'(h'^{-1}nh')n'\in HN$ and $(hn)^{-1}=h^{-1}(hn^{-1}h^{-1})\in HN$, so $HN$ is a subgroup. The
homomorphism $H\to HN/N$, $h\mapsto hN$, is surjective (as $hnN=hN$) with kernel $H\cap N$; apply (i). (iii) The map $G/N\to G/M$,
$gN\mapsto gM$, is a well-defined surjective homomorphism with kernel $M/N$; apply (i). (iv) The inverse bijection sends $\bar K\le G/N$ to
its preimage under the projection $\pi$; $\pi(\pi^{-1}(\bar K))=\bar K$ as $\pi$ is surjective, and $\pi^{-1}(\pi(K))=KN=K$ when $N\le K$.
Normality is preserved in both directions since $\pi(gKg^{-1})=\pi(g)\pi(K)\pi(g)^{-1}$.
\end{proof}

\begin{lemma}\label{lem:groups:index2}
\begin{enumerate}[label=(\roman*)]
\item A subgroup $H$ of index $2$ in $G$ is normal, and is the kernel of a surjection $G\to C_2$.
\item Let $\rho\colon G\to\Gamma$ and $\delta\colon\Gamma\to A$ be homomorphisms and $G_0=\ker(\delta\circ\rho)$. Then $\rho(G_0)=\rho(G)\cap\ker\delta$. In
particular, if $\rho\colon G\to\GL_2(F)$ and $G_0=\ker(\det\circ\rho)$, then (with $\SL_2(F)$ the group of matrices of determinant $1$,
\Cref{def:groups:gl2}) $\rho(G_0)=\rho(G)\cap\SL_2(F)$ is the determinant-one subgroup of the
image.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) The left cosets are $H$ and $G\setminus H$, and so are the right cosets. So $gH=Hg$ for every $g$, that is, $H\trianglelefteq G$; and $G/H$ has
order $2$. (ii) If $g\in G_0$, then $\rho(g)\in\rho(G)$ and $\delta(\rho(g))=1$. Conversely, if $\gamma\in\rho(G)\cap\ker\delta$, write $\gamma=\rho(g)$; then
$\delta(\rho(g))=1$, so $g\in G_0$ and $\gamma\in\rho(G_0)$.
\end{proof}

\begin{proposition}\label{prop:groups:conjclasses}
Conjugacy is an equivalence relation on $G$. For $G$ finite, the class of $g$ has $[G:C_G(g)]$ elements, and a subgroup $H$ has
exactly $[G:N_G(H)]$ conjugates $gHg^{-1}$. A subgroup is normal iff it is a union of conjugacy classes.
\end{proposition}

\begin{proof}
Reflexivity, symmetry and transitivity follow from $g=1g1^{-1}$, $h=x^{-1}(xhx^{-1})x$ and $y(xgx^{-1})y^{-1}=(yx)g(yx)^{-1}$. The map
$G\to\{xgx^{-1}\}$, $x\mapsto xgx^{-1}$, satisfies $xgx^{-1}=ygy^{-1}$ iff $y^{-1}x\in C_G(g)$ iff $xC_G(g)=yC_G(g)$; so the class is in bijection
with $G/C_G(g)$. The same argument with $x\mapsto xHx^{-1}$ and $N_G(H)$ counts the conjugates of $H$. A subgroup $N$ satisfies
$gNg^{-1}\subseteq N$ for all $g$ iff it contains the class of each of its elements; and $gNg^{-1}\subseteq N$ for all $g$ implies equality (apply it to
$g^{-1}$). (This is the special case of \Cref{thm:groups:orbitstab} for the conjugation action.)
\end{proof}

\begin{proposition}\label{prop:groups:lprimary}
Let $A$ be a finite abelian group, written additively, and $\ell$ a prime.
\begin{enumerate}[label=(\roman*)]
\item (Cauchy for abelian groups) If $\ell\mid\#A$, then $A$ has an element of order $\ell$.
\item If $\#A=mn$ with $\gcd(m,n)=1$, then $A=A[m]\oplus A[n]$, where $A[m]=\{a:ma=0\}$, and $\#A[m]=m$, $\#A[n]=n$.
\item The $\ell$-primary part $A[\ell^\infty]=\{a:\ell^ka=0\text{ for some }k\}$ equals $A[\ell^{v_\ell(\#A)}]$, has order $\ell^{v_\ell(\#A)}$, and is the unique
Sylow $\ell$-subgroup of $A$ (the unique subgroup of order $\ell^{v_\ell(\#A)}$; Sylow subgroups of arbitrary finite groups are
treated in \Cref{thm:groups:sylow}); every $\ell$-power-order subgroup lies in it.
\item $A=\bigoplus_{\ell\mid\#A}A[\ell^\infty]$, and the projection onto $A[\ell^\infty]$ is multiplication by an integer $e_\ell$ with $e_\ell\equiv1\pmod{\ell^{v_\ell(\#A)}}$
and $e_\ell\equiv0$ modulo $\#A/\ell^{v_\ell(\#A)}$.
\item The exponent of $A$ (the least $e\ge1$ with $eA=0$) divides $\#A$ and has the same prime divisors.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Induction on $\#A$. Pick $a\ne0$ of order $k$. If $\ell\mid k$, then $(k/\ell)a$ has order $\ell$. Otherwise $\ell$ divides
$\#(A/\langle a\rangle)=\#A/k<\#A$, and by induction there is $b\in A$ whose class $\bar b$ has order $\ell$. If $b$ has order $k'$, then
$k'\bar b=0$, so $\ell\mid k'$, and $(k'/\ell)b$ has order $\ell$.

(ii) Write $1=xm+yn$ with $x,y\in\Z$. For $a\in A$, $a=xma+yna$, and $n(xma)=x(mn)a=0$ by Lagrange, so $xma\in A[n]$; similarly $yna\in A[m]$.
If $a\in A[m]\cap A[n]$, then $a=xma+yna=0$. So $A=A[m]\oplus A[n]$ and $\#A[m]\cdot\#A[n]=mn$. By (i), no prime dividing $n$ divides
$\#A[m]$ (an element of prime order $p\mid n$ in $A[m]$ would lie in $A[m]\cap A[n]$, as $p\mid n$), and conversely; so $\#A[m]=m$.

(iii) Let $e=v_\ell(\#A)$ and $\#A=\ell^em$. If $\ell^ka=0$, then also $\#A\cdot a=0$, so $\gcd(\ell^k,\#A)a=0$ and $\ell^ea=0$; so $A[\ell^\infty]=A[\ell^e]$, of
order $\ell^e$ by (ii). A subgroup of $\ell$-power order consists of elements killed by a power of $\ell$, so lies in $A[\ell^\infty]$; a
subgroup of order $\ell^e$ therefore equals it.

(iv) Apply (ii) repeatedly to the factorisation of $\#A$ into prime powers. For the projector, take $e_\ell=y n$ in (ii) with
$m=\ell^{v_\ell(\#A)}$: the proof shows $a\mapsto yna$ is the projection onto $A[m]$, and $yn\equiv1\pmod m$, $yn\equiv0\pmod n$.

(v) $\#A\cdot A=0$ by Lagrange, so the exponent divides $\#A$ (write $\#A=ce+r$); every prime dividing $\#A$ is the order of some element
by (i), hence divides the exponent.
\end{proof}

\section[Products and the small groups]{Direct and semidirect products; dihedral, quaternion and semidihedral groups}\label{sec:groups:products}

This section builds the small groups that the paper names. Proposition prop:ns says that the image of $\bar\rho_{E,3}$ is
$\GL_2(\F_3)$, the normaliser $N_{ns}(\F_3)$ of a nonsplit Cartan subgroup (order $16$) or the normaliser $N_s(\F_3)$ of a split
Cartan subgroup (order $8$), and that their determinant-one subgroups are $\SL_2(\F_3)$, the quaternion group $Q_8$ and a cyclic group
of order $4$. To check such sentences one needs the groups $D_4$, $Q_8$ and $SD_{16}$ as abstract groups, together with criteria that
recognise them from a few generators and relations; these are \Cref{prop:groups:dihedral}, \Cref{prop:groups:quaternion} and
\Cref{prop:groups:semidihedral}, all resting on the elementary \Cref{lem:groups:normalform}. Semidirect products also describe the
Borel subgroup, the normalisers of Cartan subgroups and, in \Cref{ch:local}, the tame Galois group of $\Q_p$.

\begin{proposition}[Direct products]\label{prop:groups:directproduct}
The \emph{direct product} $G\times H$ of two groups is the set of pairs with $(g,h)(g',h')=(gg',hh')$. If $H$ and $K$ are normal subgroups of a
group $G$ with $H\cap K=1$ and $HK=G$, then elements of $H$ commute with elements of $K$, and $(h,k)\mapsto hk$ is an isomorphism
$H\times K\to G$.
\end{proposition}

\begin{proof}
For $h\in H$ and $k\in K$ the commutator $hkh^{-1}k^{-1}$ equals $h\cdot(kh^{-1}k^{-1})\in H$, as $H$ is normal, and equals $(hkh^{-1})\cdot k^{-1}\in K$, as $K$ is
normal; so it lies in $H\cap K=1$ and $hk=kh$. Hence $(h,k)(h',k')=(hh',kk')\mapsto hh'kk'=hkh'k'$, and the map is a homomorphism. If $hk=1$, then
$h=k^{-1}\in H\cap K$, so $h=k=1$: the map is injective. It is surjective because $HK=G$.
\end{proof}

\begin{proposition}[Chinese remainder theorem for groups]\label{prop:groups:crtgroups}
Let $m,n\ge1$. Then $C_m\times C_n$ is cyclic (and so $\simeq C_{mn}$) if and only if $\gcd(m,n)=1$. Consequently
$\Z/n\simeq\prod_{p\mid n}\Z/p^{v_p(n)}$ as groups, and a finite cyclic group is the direct product of its $\ell$-primary parts.
\end{proposition}

\begin{proof}
Let $s$, $t$ generate $C_m$, $C_n$. The order of $(s^a,t^b)$ is the least common multiple of the orders of $s^a$ and $t^b$, which divides
$\lcm(m,n)$. If $\gcd(m,n)=1$, the element $(s,t)$ has order $\lcm(m,n)=mn=\#(C_m\times C_n)$ and generates. If $d=\gcd(m,n)>1$, every
element has order dividing $\lcm(m,n)=mn/d<mn$, so no element generates. The decomposition of $\Z/n$ follows by induction on the number
of prime factors of $n$, splitting off one prime power at a time. For a cyclic group $C$ of order $n$, the factor $\Z/p^{v_p(n)}$ is
the unique subgroup of order $p^{v_p(n)}$ (\Cref{thm:groups:cyclic}(i)), which is $C[p^\infty]$ by \Cref{prop:groups:lprimary}(iii).
\end{proof}

\begin{definition}[Semidirect products]\label{def:groups:semidirect}
Let $N$ and $H$ be groups and $\theta\colon H\to\Aut(N)$ a homomorphism. The \emph{semidirect product} $N\rtimes_\theta H$ is the set $N\times H$
with the product
\[
(n,h)(n',h')=\bigl(n\,\theta(h)(n'),\ hh'\bigr).
\]
\end{definition}

The product is associative: both $\bigl((n,h)(n',h')\bigr)(n'',h'')$ and $(n,h)\bigl((n',h')(n'',h'')\bigr)$ equal
$\bigl(n\,\theta(h)(n')\,\theta(hh')(n''),\ hh'h''\bigr)$, because $\theta(h)\bigl(n'\theta(h')(n'')\bigr)=\theta(h)(n')\,\theta(h)\theta(h')(n'')$ and $\theta$ is a
homomorphism. The identity is $(1,1)$ and the inverse of $(n,h)$ is $\bigl(\theta(h^{-1})(n^{-1}),h^{-1}\bigr)$. The maps $n\mapsto(n,1)$ and $h\mapsto(1,h)$
identify $N$ and $H$ with subgroups, and
\[
(1,h)(n,1)(1,h)^{-1}=\bigl(\theta(h)(n),h\bigr)(1,h^{-1})=\bigl(\theta(h)(n),1\bigr),
\]
so that $hnh^{-1}=\theta(h)(n)$, as in convention (C2). The projection $(n,h)\mapsto h$ is a homomorphism with kernel $N$, so $N$ is normal.

\begin{proposition}[Recognising semidirect products]\label{prop:groups:semidirect}
\begin{enumerate}[label=(\roman*)]
\item Let $N\trianglelefteq G$ and $H\le G$ with $N\cap H=1$ and $NH=G$, and let $\theta(h)(n)=hnh^{-1}$. Then $(n,h)\mapsto nh$ is an isomorphism
$N\rtimes_\theta H\to G$.
\item A short exact sequence $1\to N\to G\xrightarrow{\pi}Q\to1$ \emph{splits}, that is, has a homomorphism $s\colon Q\to G$ with $\pi\circ s=\id$, if and only
if $G\simeq N\rtimes_\theta Q$ for some $\theta$, compatibly with the maps from $N$ and to $Q$.
\item In $N\rtimes_\theta H$ the subgroup $H$ is normal if and only if $\theta$ is trivial, in which case $N\rtimes_\theta H=N\times H$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $nh\cdot n'h'=n(hn'h^{-1})hh'$, so the map is a homomorphism. If $nh=n'h'$, then $n'^{-1}n=h'h^{-1}\in N\cap H=1$, so it is injective; it is
surjective since $NH=G$. (ii) Given $s$, put $H=s(Q)$. If $s(x)\in N$, then $x=\pi(s(x))=1$, so $N\cap H=1$; and every $g$ equals
$\bigl(g\,s(\pi(g))^{-1}\bigr)\,s(\pi(g))$ with the first factor in $\ker\pi=N$, so $NH=G$; now apply (i), noting $s\colon Q\to H$ is an isomorphism.
Conversely $q\mapsto(1,q)$ is a section of $N\rtimes_\theta Q\to Q$. (iii) $(n,1)(1,h)(n,1)^{-1}=\bigl(n\,\theta(h)(n^{-1}),h\bigr)$ lies in $H$ for all $n$
iff $\theta(h)(n)=n$ for all $n$. If $\theta$ is trivial the product formula is the componentwise one.
\end{proof}

\begin{example}\label{ex:groups:semidirectexamples}
The following semidirect products occur in this chapter and later in the book.
\begin{enumerate}[label=(\alph*)]
\item $D_n=C_n\rtimes C_2$, with $C_2$ acting by inversion (\Cref{def:groups:dihedral}).
\item $\GL_2(F)=\SL_2(F)\rtimes\{\mathrm{diag}(1,d):d\in F^\times\}$ (\Cref{prop:groups:gl2}(ii)), and $\mathrm B=\mathrm U\rtimes\mathrm T$ for the upper
triangular group (\Cref{prop:groups:borel}).
\item $N_s=C_s\rtimes\langle w\rangle$ and $N_{ns}=C_{ns}\rtimes\langle\sigma\rangle$ for the normalisers of Cartan subgroups (\Cref{prop:groups:cartan}), and
$\SL_2(\F_3)=Q_8\rtimes C_3$ (\Cref{prop:groups:sl23}).
\item $S_4=V_4\rtimes S_3$, with $V_4=\{1,(12)(34),(13)(24),(14)(23)\}$ normal and $S_3$ the stabiliser of the point $4$: they meet trivially and
$\#V_4\cdot\#S_3=24$.
\item (Foretaste of \Cref{ch:local}.) The Galois group of the maximal tamely ramified extension of $\Q_p$ is a semidirect product
$\bigl(\prod_{\ell\ne p}\Z_\ell(1)\bigr)\rtimes\widehat\Z$, in which a Frobenius element acts on the tame inertia group by
$\Frob\,t\,\Frob^{-1}=t^p$. The paper uses this in the proofs of Lemma lem:tamfactor and Lemma lem:IV, in the form
$\Frob_q\,\sigma\,\Frob_q^{-1}=\sigma^q$ for an element $\sigma$ of tame inertia.
\end{enumerate}
\end{example}

The groups $D_n$, $Q_8$ and $SD_{16}$ each consist of the powers of one element $a$ and one coset of them. The following lemma, used
three times below, says that such a group is determined by three numbers.

\begin{lemma}[Normal forms]\label{lem:groups:normalform}
Let $n\ge1$, $k$ and $c$ be integers with $k^2\equiv1\pmod n$ and $kc\equiv c\pmod n$. Let $G$ be a group with elements $a$, $b$ such that
$a^n=1$, $bab^{-1}=a^k$, $b^2=a^c$, and such that the $2n$ elements $a^mb^e$ ($0\le m<n$, $e\in\{0,1\}$) are distinct and exhaust $G$.
Then the product in $G$ is given by
\[
a^mb^e\cdot a^{m'}b^{e'}=\begin{cases}a^{m+k^em'}b^{e+e'}&\text{if }e+e'\le1,\\ a^{m+km'+c}&\text{if }e=e'=1.\end{cases}
\]
In particular, if $G'$, $a'$, $b'$ satisfy the same hypotheses with the same $(n,k,c)$, then $a^mb^e\mapsto a'^mb'^e$ is an isomorphism $G\to G'$.
\end{lemma}

\begin{proof}
From $bab^{-1}=a^k$ we get $ba^{m'}b^{-1}=a^{km'}$, so $b^ea^{m'}=a^{k^em'}b^e$, and $a^mb^ea^{m'}b^{e'}=a^{m+k^em'}b^eb^{e'}$; when $e=e'=1$,
$b^eb^{e'}=b^2=a^c$. Since exponents of $a$ only matter modulo $n$, the right-hand side is computed from $(n,k,c)$ alone, and the
bijection $a^mb^e\mapsto a'^mb'^e$ respects products. (The congruences on $k$ and $c$ are forced: $a=b^2ab^{-2}=a^{k^2}$ and $b$ commutes with
$b^2=a^c$; they are not used in the proof.)
\end{proof}

\begin{definition}[Dihedral groups]\label{def:groups:dihedral}
For $n\ge2$ the \emph{dihedral group} of order $2n$ is $D_n=\Z/n\rtimes_\theta\Z/2$, with $\theta(1)$ the inversion $x\mapsto-x$. Writing
$r=(1,0)$ and $s=(0,1)$, we have $r^n=s^2=1$ and $srs^{-1}=r^{-1}$, and $D_n=\{r^ms^e\}$. For $n\ge3$, $D_n$ is the group of symmetries of a
regular $n$-gon in the plane, $r$ being the rotation by $2\pi/n$ and $s$ a reflection (the familiar picture from Riley-Hobson-Bence).
\end{definition}

\begin{proposition}\label{prop:groups:dihedral}
\begin{enumerate}[label=(\roman*)]
\item (Recognition) Let $G$ be a group of order $2n$ generated by an element $r$ of order $n$ and an element $s$ of order $2$ with
$srs^{-1}=r^{-1}$. Then $G\simeq D_n$, with $r\mapsto(1,0)$ and $s\mapsto(0,1)$.
\item Every element of $D_n$ outside $\langle r\rangle$ has order $2$.
\item For $n\ge3$ the centre of $D_n$ is trivial if $n$ is odd and is $\{1,r^{n/2}\}$ if $n$ is even.
\item $D_2\simeq V_4$, $D_3\simeq S_3$ and $D_6\simeq C_2\times S_3$.
\item $D_4$ has five elements of order $2$ (namely $r^2$, $s$, $rs$, $r^2s$, $r^3s$) and two of order $4$ (namely $r$, $r^3$).
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Since $srs^{-1}=r^{-1}$, both generators normalise $\langle r\rangle$, which is therefore normal, and $\langle r\rangle\langle s\rangle$ is a subgroup
(\Cref{thm:groups:isothms}(ii)) containing $r$ and $s$, hence equal to $G$. Its order is $n\cdot2/\#(\langle r\rangle\cap\langle s\rangle)$, which must be $2n$, so
$s\notin\langle r\rangle$ and the elements $r^ms^e$ are distinct and exhaust $G$. Apply \Cref{lem:groups:normalform} with $(n,k,c)=(n,-1,0)$ to $G$ and
to $D_n$. (ii) $(r^ms)^2=r^m(sr^ms^{-1})=r^mr^{-m}=1$, and $r^ms\ne1$. (iii) $r^m$ commutes with $s$ iff $r^m=r^{-m}$ iff $n\mid2m$, which gives
$m=0$, or $m=n/2$ when $n$ is even; $r^m$ commutes with $r$. An element $r^ms$ is not central: $r(r^ms)r^{-1}=r^{m+2}s\ne r^ms$ as $n\ge3$.
(iv) $D_2$ has order $4$ and all its nontrivial elements have order $2$ by (ii) and $r^2=1$; so it is $V_4$ (it is abelian, being generated by
two commuting involutions, as $srs^{-1}=r^{-1}=r$). In $S_3$, $r=(123)$ and $s=(12)$ satisfy $srs^{-1}=(213)=r^{-1}$, so (i) gives $S_3\simeq D_3$. In $D_6$, the
centre $Z=\{1,r^3\}$ and the subgroup $K=\langle r^2,s\rangle=\{1,r^2,r^4,s,r^2s,r^4s\}$ are normal ($K$ has index $2$), $Z\cap K=1$ and $ZK=D_6$; so
$D_6\simeq C_2\times K$ by \Cref{prop:groups:directproduct}, and $K\simeq D_3\simeq S_3$ by (i). (v) follows from (ii) and the orders of the powers of $r$.
\end{proof}

\begin{definition}[The quaternion group]\label{def:groups:quaternion}
Let $\mathbf i=\sqrt{-1}\in\C$. In $\GL_2(\C)$ put
\[
1=I,\qquad i=\Mtwo{\mathbf i}00{-\mathbf i},\qquad j=\Mtwo01{-1}0,\qquad k=ij=\Mtwo0{\mathbf i}{\mathbf i}0 .
\]
The \emph{quaternion group} is $Q_8=\{\pm1,\pm i,\pm j,\pm k\}$. Direct multiplication gives $i^2=j^2=k^2=ijk=-1$, $ji=-k$, and
$jij^{-1}=-i=i^{-1}$; so $Q_8$ is a subgroup of order $8$, and $Q_8=\{i^mj^e:0\le m<4,\ e\in\{0,1\}\}$ with $i^4=1$, $jij^{-1}=i^{-1}$, $j^2=i^2$.
(These are Hamilton's quaternion units.)
\end{definition}

\begin{proposition}\label{prop:groups:quaternion}
\begin{enumerate}[label=(\roman*)]
\item $-1$ is the unique element of order $2$ of $Q_8$, and the six elements $\pm i,\pm j,\pm k$ have order $4$.
\item $Z(Q_8)=[Q_8,Q_8]=\{\pm1\}$, and $Q_8/\{\pm1\}\simeq V_4$.
\item Every subgroup of $Q_8$ is normal, and every nontrivial subgroup contains $-1$. In particular $Q_8$ is not a semidirect product of two
nontrivial subgroups.
\item $Q_8\not\simeq D_4$.
\item (Recognition) A non-abelian group of order $8$ with a unique element of order $2$ is isomorphic to $Q_8$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Each of $\pm i,\pm j,\pm k$ squares to $-1$. (ii) $-1=-I$ is central; $i$ and $j$ do not commute ($ji=-ij$), and similarly no element
other than $\pm1$ is central (each of $\pm i$, $\pm j$, $\pm k$ anticommutes with one of the others). The commutator $iji^{-1}j^{-1}=ij(-i)(-j)=(ij)^2=k^2=-1$,
so $-1\in[Q_8,Q_8]$. The quotient $Q_8/\{\pm1\}$ has order $4$ and every element squares to $1$ there, since $x^2\in\{\pm1\}$; so it is $V_4$, which is
abelian, and $[Q_8,Q_8]\le\{\pm1\}$. (iii) Every $x\ne1$ satisfies $x=-1$ or $x^2=-1$, so a nontrivial subgroup contains $-1$. Subgroups of
order $1$, $8$ are normal, $\{\pm1\}$ is central, and subgroups of order $4$ have index $2$ (\Cref{lem:groups:index2}). Two nontrivial subgroups
meet in $\{\pm1\}$ at least, so $Q_8$ is not $N\rtimes H$ with $N,H\ne1$. (iv) $D_4$ has five elements of order $2$ (\Cref{prop:groups:dihedral}(v)),
$Q_8$ one. (v) Let $G$ be non-abelian of order $8$ with unique involution $z$. No element has order $8$ ($G$ would be cyclic), so every
element other than $1$, $z$ has order $4$. Choose $a$ of order $4$; then $a^2$ has order $2$, so $a^2=z$. Choose $b\notin\langle a\rangle$; it has order $4$ and
$b^2=z=a^2$. The subgroup $\langle a\rangle$ has index $2$, hence is normal, so $bab^{-1}$ is an element of order $4$ of $\langle a\rangle$, that is $a$ or $a^{-1}$.
If $bab^{-1}=a$, then $G=\langle a\rangle\cup\langle a\rangle b$ is abelian, a contradiction. So $bab^{-1}=a^{-1}$, and \Cref{lem:groups:normalform} with
$(n,k,c)=(4,-1,2)$, applied to $(G,a,b)$ and $(Q_8,i,j)$, gives $G\simeq Q_8$.
\end{proof}

\begin{definition}[The semidihedral group]\label{def:groups:semidihedral}
The \emph{semidihedral group} of order $16$ is $SD_{16}=\Z/8\rtimes_\theta\Z/2$ with $\theta(1)$ the automorphism $x\mapsto3x$ (of order $2$, as
$9\equiv1\pmod8$). Writing $a=(1,0)$, $x=(0,1)$, we have $a^8=x^2=1$, $xax^{-1}=a^3$ and $SD_{16}=\{a^mx^e\}$.
\end{definition}

\begin{proposition}\label{prop:groups:semidihedral}
\begin{enumerate}[label=(\roman*)]
\item $(a^mx)^2=a^{4m}$; so $a^mx$ has order $2$ for $m$ even and order $4$ for $m$ odd.
\item $SD_{16}$ has $1$, $5$, $6$, $4$ elements of orders $1$, $2$, $4$, $8$, and $Z(SD_{16})=\{1,a^4\}$.
\item The subgroups of index $2$ of $SD_{16}$ are exactly three: $\langle a\rangle\simeq C_8$, $\langle a^2,x\rangle\simeq D_4$ and $\langle a^2,ax\rangle\simeq Q_8$.
\item (Recognition) A group of order $16$ generated by an element $a$ of order $8$ and an element $x$ of order $2$ with $xax^{-1}=a^3$ is
isomorphic to $SD_{16}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $(a^mx)^2=a^m(xa^mx^{-1})=a^ma^{3m}=a^{4m}$, which is $1$ for $m$ even and $a^4\ne1$ for $m$ odd. (ii) In $\langle a\rangle$ there are $1,1,2,4$
elements of orders $1,2,4,8$; in the coset $\langle a\rangle x$ there are four of order $2$ and four of order $4$ by (i). An element $a^m$ is central iff
$xa^mx^{-1}=a^{3m}$ equals $a^m$, iff $a^{2m}=1$, iff $m\in\{0,4\}$; no $a^mx$ is central, since $a(a^mx)a^{-1}=a^{m+1}a^{-3}x=a^{m-2}x$.
(iii) A subgroup $K$ of index $2$ is normal with quotient of order $2$, so it contains every square; the squares include $a^2$, so
$K\supseteq\langle a^2\rangle$. The subgroup $\langle a^2\rangle$ of order $4$ is normal ($xa^2x^{-1}=a^6$), and $SD_{16}/\langle a^2\rangle$ has order $4$ and is generated
by the images of $a$ and $x$, which commute ($xax^{-1}a^{-1}=a^2$) and have order dividing $2$; so it is $V_4$, which has exactly three subgroups of
order $2$. By the correspondence theorem (\Cref{thm:groups:isothms}(iv)) there are exactly three subgroups of index $2$, namely $\langle a\rangle$,
$\langle a^2,x\rangle$ and $\langle a^2,ax\rangle$. The second is dihedral by \Cref{prop:groups:dihedral}(i): it has order $8$, $a^2$ has order $4$, $x$ has order $2$, and
$xa^2x^{-1}=a^6=a^{-2}$. The third consists of $a^{2m}$ and $a^{2m+1}x$; by (i) the latter have order $4$, so $a^4$ is its unique involution, and it is
non-abelian ($(ax)a^2(ax)^{-1}=a\cdot a^6\cdot a^{-1}=a^{-2}\ne a^2$); so it is $Q_8$ by \Cref{prop:groups:quaternion}(v). (iv) As in
\Cref{prop:groups:dihedral}(i), $\langle a\rangle$ is normal, $G=\langle a\rangle\langle x\rangle$, and the order $16$ forces $x\notin\langle a\rangle$; apply
\Cref{lem:groups:normalform} with $(n,k,c)=(8,3,0)$.
\end{proof}

\begin{remark}\label{rem:groups:order16}
There are four non-abelian groups of order $16$ with a cyclic subgroup $\langle a\rangle$ of order $8$ and an element $x\notin\langle a\rangle$ with $xax^{-1}=a^k$,
$x^2=a^c$: the dihedral group $D_8$ ($k=-1$, $c=0$), $SD_{16}$ ($k=3$, $c=0$), the generalised quaternion group $Q_{16}$ ($k=-1$, $c=4$) and the
modular group $M_{16}$ ($k=5$, $c=0$). In each case $(a^mx)^2=a^{m(1+k)+c}$, and counting as in \Cref{prop:groups:semidihedral} gives the numbers of
elements of orders $1,2,4,8$:
\begin{center}
\begin{tabular}{lcccc}
\toprule
group & order $1$ & order $2$ & order $4$ & order $8$\\
\midrule
$D_8$ & 1 & 9 & 2 & 4\\
$SD_{16}$ & 1 & 5 & 6 & 4\\
$Q_{16}$ & 1 & 1 & 10 & 4\\
$M_{16}$ & 1 & 3 & 4 & 8\\
\bottomrule
\end{tabular}
\end{center}
So these four groups are pairwise non-isomorphic, and a group of order $16$ is recognised as $SD_{16}$ by its element orders among them.
We use only $SD_{16}$, through \Cref{prop:groups:semidihedral}(iv).
\end{remark}

\begin{theorem}[Groups of order 8]\label{thm:groups:order8}
Every group of order $8$ is isomorphic to exactly one of $C_8$, $C_4\times C_2$, $C_2^3$, $D_4$, $Q_8$.
\end{theorem}

\begin{proof}
If $G$ is abelian, \Cref{thm:groups:finiteabelian} (proved in \Cref{sec:groups:abelian}, independently of this theorem) gives exactly the first
three, distinguished by the orders $\#G[2]=2,4,8$ of their subgroups of elements killed by $2$. Let $G$ be non-abelian. If every
element had order at most $2$, then $G$ would be abelian ($ab=(ab)^{-1}=b^{-1}a^{-1}=ba$); and no element has order $8$. So there is $r$ of order $4$,
and $\langle r\rangle$ has index $2$, hence is normal. Let $s\notin\langle r\rangle$. Then $srs^{-1}\in\{r,r^{-1}\}$ (the elements of order $4$ of $\langle r\rangle$), and
$srs^{-1}=r$ would make $G=\langle r,s\rangle$ abelian; so $srs^{-1}=r^{-1}$. Next $s^2\in\langle r\rangle$ (the quotient has order $2$) and $s^2$ commutes with $s$, so
$s^2=s(s^2)s^{-1}$, which is $(s^2)^{-1}$ if $s^2\in\{r,r^{-1}\}$; that would force $r^2=1$. So $s^2\in\{1,r^2\}$. If $s^2=1$, then $G\simeq D_4$ by
\Cref{prop:groups:dihedral}(i); if $s^2=r^2$, then $G\simeq Q_8$ by \Cref{lem:groups:normalform} with $(n,k,c)=(4,-1,2)$. The five groups are pairwise
non-isomorphic: the three abelian ones are distinguished by $\#G[2]$, and $D_4\not\simeq Q_8$ by \Cref{prop:groups:quaternion}(iv).
\end{proof}

The words ``dihedral'' and ``quaternion'' in the proof of Proposition prop:ns name isomorphism types. In
\Cref{sec:groups:gl23} they are identified by the recognition criteria: $N_s(\F_3)\simeq D_4$ by \Cref{prop:groups:dihedral}(i), the
determinant-one subgroup of $N_{ns}(\F_3)$ is $Q_8$ by \Cref{prop:groups:quaternion}(v), and $N_{ns}(\F_3)\simeq SD_{16}$ by
\Cref{prop:groups:semidihedral}(iv). \Cref{thm:groups:order8} is context; nothing in the paper depends on it.
\end{paperlink}

\section[Group actions, orbit-stabiliser, \texorpdfstring{$p$}{p}-groups, Sylow theorems]{Group actions, orbit-stabiliser, \texorpdfstring{$p$}{p}-groups and the Sylow theorems}\label{sec:groups:actions}

Groups enter the paper through their actions: the Galois group acts on the torsion points of $E$, the decomposition and inertia groups are
stabilisers (\Cref{ch:ant}, \Cref{ch:local}), and the four roots of the $3$-division polynomial correspond to the four lines of
$E[3]\simeq\F_3^2$, permuted by the image of $\bar\rho_{E,3}$. The last sentence of Proposition prop:ns, on the factorisation of that
polynomial, is a statement about the orbits of a subgroup of $\GL_2(\F_3)$ on these four lines, and \Cref{prop:groups:suborbits} is the
tool that compares the orbits of a group and of a subgroup of index $2$. The Sylow theorems are used to locate the $2$-subgroups and the
$3$-subgroups of $\GL_2(\F_3)$ in \Cref{sec:groups:gl23}.

\begin{definition}[Group actions]\label{def:groups:action}
A (left) \emph{action} of a group $G$ on a set $X$ is a map $G\times X\to X$, $(g,x)\mapsto gx$, with $1x=x$ and $g(hx)=(gh)x$. Equivalently, it is a
homomorphism $G\to\operatorname{Sym}(X)$ to the group of bijections of $X$ (the \emph{permutation representation}). Its kernel is the kernel of this
homomorphism; the action is \emph{faithful} if the kernel is trivial. The \emph{orbit} of $x$ is $Gx=\{gx:g\in G\}$, the \emph{stabiliser} of $x$ is
$G_x=\{g:gx=x\}$, a subgroup, and the \emph{fixed set} is $X^G=\{x:gx=x\ \forall g\}$; for one element $g$ we write $X^g$ for its fixed points. The action is
\emph{transitive} if there is one orbit. An action restricts to every subgroup $H\le G$. Examples: $G$ acts on $G/H$ by $g\cdot xH=gxH$; $G$ acts on itself
and on the set of its subgroups by conjugation, with stabilisers $C_G(g)$ and $N_G(H)$; $\GL_2(F)$ acts on $F^2$, on $F^2\setminus\{0\}$ and on
the set $\PP^1(F)$ of lines (\Cref{prop:groups:projline}).
\end{definition}

\begin{theorem}[Orbit-stabiliser]\label{thm:groups:orbitstab}
Let $G$ act on $X$ and let $x\in X$. Then $gG_x\mapsto gx$ is a well-defined bijection $G/G_x\to Gx$, compatible with the actions of $G$; so
$\#Gx=[G:G_x]$ when $G$ is finite. Moreover $G_{gx}=gG_xg^{-1}$. The orbits partition $X$, and if $X$ is finite and $x_1,\dots,x_r$ are representatives
of the orbits, then $\#X=\sum_{i=1}^r[G:G_{x_i}]$.
\end{theorem}

\begin{proof}
$gx=hx$ iff $h^{-1}g\in G_x$ iff $gG_x=hG_x$: so the map is well defined and injective; it is surjective by definition of $Gx$, and
$h\cdot gG_x=hgG_x\mapsto hgx$. If $h\in G_x$, then $(ghg^{-1})(gx)=gx$, so $gG_xg^{-1}\subseteq G_{gx}$; applying this to $g^{-1}$ and $gx$ gives the
reverse inclusion. The relation ``$y\in Gx$'' is an equivalence relation ($x=1x$; $y=gx\Rightarrow x=g^{-1}y$; $z=hy$, $y=gx\Rightarrow z=hgx$), whose
classes are the orbits. Summing the orbit sizes gives the last formula.
\end{proof}

\begin{proposition}[Orbits of a subgroup]\label{prop:groups:suborbits}
Let $G$ act transitively on a finite set $X$, let $x\in X$ and let $H\le G$.
\begin{enumerate}[label=(\roman*)]
\item The map $HgG_x\mapsto H\cdot gx$ is a bijection from the double cosets $H\backslash G/G_x$ to the $H$-orbits on $X$. In particular $H$ is transitive
on $X$ if and only if $G=HG_x$.
\item If $H\trianglelefteq G$, then $G$ permutes the $H$-orbits transitively; all $H$-orbits have the same size, which divides $\#X$.
\item If $[G:H]=2$, then either $H$ is transitive on $X$, or $X$ is the union of exactly two $H$-orbits, each of size $\#X/2$, and every element of
$G\setminus H$ interchanges them.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Every element of $X$ is $gx$ for some $g$. Now $Hgx=Hg'x$ iff $g'x=hgx$ for some $h\in H$ iff $g^{-1}h^{-1}g'\in G_x$ for some $h$ iff $g'\in HgG_x$. So the
map is well defined and injective, and it is clearly surjective. There is one double coset iff $HG_x=G$ (the double coset of $1$ is $HG_x$).
(ii) For $g\in G$ and $y\in X$, $g(Hy)=gHg^{-1}gy=H(gy)$; so $g$ maps $H$-orbits to $H$-orbits, bijectively (with inverse $g^{-1}$). Given two orbits
$Hy$, $Hy'$, choose $g$ with $gy=y'$ (transitivity of $G$); then $g(Hy)=Hy'$. Hence all orbits have the size of one of them, say $s$, and
$\#X=s\cdot(\text{number of orbits})$. (iii) $H$ is normal by \Cref{lem:groups:index2}. The group $G$ acts transitively on the set $\Omega$ of $H$-orbits by (ii),
and $H$ acts trivially on $\Omega$; so the action factors through $G/H$, of order $2$, and $\#\Omega\in\{1,2\}$. If $\#\Omega=2$, each orbit has size $\#X/2$ by
(ii); the stabiliser of one $H$-orbit is a subgroup containing $H$ and different from $G$ (the action on $\Omega$ is transitive), hence equal to $H$, so every
element of $G\setminus H$ moves it to the other one.
\end{proof}

\begin{lemma}[Orbit counting]\label{lem:groups:burnside}
Let a finite group $G$ act on a finite set $X$. The number of orbits is $\dfrac1{\#G}\displaystyle\sum_{g\in G}\#X^g$.
\end{lemma}

\begin{proof}
Count the pairs $(g,x)$ with $gx=x$ in two ways: $\sum_g\#X^g=\sum_x\#G_x$. By \Cref{thm:groups:orbitstab}, $\#G_x=\#G/\#Gx$, so the elements of one
orbit $O$ contribute $\#O\cdot\#G/\#O=\#G$, and the sum is $\#G$ times the number of orbits.
\end{proof}

\begin{theorem}[Class equation; $p$-groups]\label{thm:groups:classeq}
Let $G$ be finite. Then $\#G=\#Z(G)+\sum_i[G:C_G(g_i)]$, the sum over representatives $g_i$ of the conjugacy classes with more than one element.
Let $p$ be a prime.
\begin{enumerate}[label=(\roman*)]
\item If a finite $p$-group $P$ acts on a finite set $X$, then $\#X\equiv\#X^P\pmod p$.
\item A nontrivial finite $p$-group has nontrivial centre.
\item A group of order $p^2$ is abelian.
\item A group of order $p^n$ has a normal subgroup of order $p^k$ for every $0\le k\le n$, and every subgroup of index $p$ in a finite $p$-group is normal.
\end{enumerate}
\end{theorem}

\begin{proof}
The class equation is \Cref{thm:groups:orbitstab} for the conjugation action: the fixed points are the central elements, and the class of $g_i$ has
$[G:C_G(g_i)]$ elements (\Cref{prop:groups:conjclasses}). (i) Each orbit has size $[P:P_x]$, a power of $p$, which is $1$ exactly for the fixed points; so
$\#X-\#X^P$ is a sum of positive powers of $p$. (ii) Apply (i) to the conjugation action of $P$ on itself: $\#Z(P)\equiv\#P\equiv0\pmod p$, and
$\#Z(P)\ge1$, so $\#Z(P)\ge p$. (iii) First, if $G/Z(G)$ is cyclic, generated by the class of $g$, then every element is $g^mz$ with $z$ central, and
any two such elements commute; so $G$ is abelian. If $\#G=p^2$, then $\#Z(G)\in\{p,p^2\}$ by (ii); if it were $p$, then $G/Z(G)$ would have order $p$,
hence be cyclic, and $G$ would be abelian, so $Z(G)=G$, a contradiction. (iv) Induction on $n$; the case $n=0$ is empty. For $n\ge1$, $Z(G)$ is a
nontrivial abelian $p$-group, so it has an element $z$ of order $p$ (\Cref{prop:groups:lprimary}(i)), and $\langle z\rangle$ is normal, being central. For
$k\ge1$, by induction $G/\langle z\rangle$ (order $p^{n-1}$) has a normal subgroup of order $p^{k-1}$, whose preimage is normal of order $p^k$
(\Cref{thm:groups:isothms}(iv)). Finally let $H\le P$ have index $p$. Then $H$ acts on $P/H$ by left multiplication; by (i), the number of fixed cosets is
$\equiv p\equiv0\pmod p$, and the coset $H$ is fixed, so there are at least $p$ fixed cosets, that is, all of them. So $hgH=gH$ for all $h\in H$, $g\in P$, that
is, $g^{-1}Hg\subseteq H$ for all $g$, and $H$ is normal.
\end{proof}

\begin{theorem}[Cauchy]\label{thm:groups:cauchy}
If a prime $p$ divides the order of a finite group $G$, then $G$ has an element of order $p$.
\end{theorem}

\begin{proof}
(McKay's argument.) Let $X=\{(g_1,\dots,g_p)\in G^p:g_1g_2\cdots g_p=1\}$. The first $p-1$ entries may be chosen freely and determine the last, so
$\#X=\#G^{p-1}\equiv0\pmod p$. The cyclic group $C_p$ acts on $X$ by rotation $(g_1,\dots,g_p)\mapsto(g_2,\dots,g_p,g_1)$; this preserves $X$, since
$g_2\cdots g_pg_1=g_1^{-1}(g_1\cdots g_p)g_1=1$. The fixed points are the constant tuples $(g,\dots,g)$ with $g^p=1$. By
\Cref{thm:groups:classeq}(i) their number is $\equiv\#X\equiv0\pmod p$; it is at least $1$ (take $g=1$), hence at least $p$. So some $g\ne1$ has
$g^p=1$, and it has order $p$.
\end{proof}

\begin{theorem}[Sylow]\label{thm:groups:sylow}
Let $G$ be a finite group of order $p^am$ with $p$ prime and $p\nmid m$. A subgroup of order $p^a$ is a \emph{Sylow $p$-subgroup}.
\begin{enumerate}[label=(\roman*)]
\item $G$ has a Sylow $p$-subgroup.
\item Every $p$-subgroup of $G$ is contained in a Sylow $p$-subgroup, and any two Sylow $p$-subgroups are conjugate.
\item The number $n_p$ of Sylow $p$-subgroups equals $[G:N_G(P)]$ for any Sylow $p$-subgroup $P$; it divides $m$, and $n_p\equiv1\pmod p$.
\item A Sylow $p$-subgroup $P$ is normal if and only if $n_p=1$; a normal Sylow $p$-subgroup contains every element of $p$-power order of $G$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) (Wielandt's argument.) First, $\binom{p^am}{p^a}\equiv m\pmod p$: in $\F_p[X]$ we have $(1+X)^p=1+X^p$, since $p$ divides $\binom pi$ for
$0<i<p$; hence $(1+X)^{p^am}=(1+X^{p^a})^m$, and comparing the coefficients of $X^{p^a}$ gives the claim. Let $\Omega$ be the set of subsets of $G$ with
$p^a$ elements, on which $G$ acts by left multiplication. As $\#\Omega\equiv m\not\equiv0\pmod p$, some orbit $GS$ has size prime to $p$; then
$\#G_S=\#G/\#GS$ is divisible by $p^a$. On the other hand $G_SS=S$, so $S$ is a union of right cosets $G_Ss$, and $\#G_S\le\#S=p^a$. Hence $G_S$ is a
Sylow $p$-subgroup. (ii) Let $Q$ be a $p$-subgroup and $P$ a Sylow $p$-subgroup. Then $Q$ acts on $G/P$ by left multiplication, and $\#(G/P)=m\not\equiv0
\pmod p$; by \Cref{thm:groups:classeq}(i) there is a fixed coset $gP$. Then $QgP=gP$, so $g^{-1}Qg\subseteq P$ and $Q\subseteq gPg^{-1}$, which is a Sylow
$p$-subgroup. If $Q$ is itself Sylow, comparing orders gives $Q=gPg^{-1}$. (iii) By (ii), $G$ acts transitively by conjugation on the set $\mathrm{Syl}_p$ of
Sylow $p$-subgroups, with stabiliser $N_G(P)$; so $n_p=[G:N_G(P)]$, which divides $[G:P]=m$ since $P\le N_G(P)$. Let $P$ act on $\mathrm{Syl}_p$ by
conjugation. If $Q$ is fixed, then $P\le N_G(Q)$, and $P$, $Q$ are Sylow $p$-subgroups of $N_G(Q)$, in which $Q$ is normal; by (ii) applied to the group $N_G(Q)$ they are
conjugate in $N_G(Q)$, and $Q$ is normal there, so $P=Q$. So $P$ has exactly one fixed point, and $n_p\equiv1\pmod p$ by \Cref{thm:groups:classeq}(i). (iv) The first statement is
$n_p=[G:N_G(P)]$. If $P$ is normal and $g$ has $p$-power order, then $\langle g\rangle$ lies in some Sylow $p$-subgroup by (ii), which is $P$.
\end{proof}

\begin{example}\label{ex:groups:sylowsmall}
(a) A group of order $15$ is cyclic: $n_3$ divides $5$ and is $\equiv1\pmod3$, so $n_3=1$; $n_5$ divides $3$ and is $\equiv1\pmod5$, so $n_5=1$. The two
Sylow subgroups $P_3\simeq C_3$, $P_5\simeq C_5$ are normal, meet trivially and $\#P_3P_5=15$; so $G\simeq C_3\times C_5\simeq C_{15}$ by
\Cref{prop:groups:directproduct} and \Cref{prop:groups:crtgroups}.

(b) In $S_4$ (order $24=2^3\cdot3$): the eight $3$-cycles lie in four subgroups of order $3$, so $n_3=4$; the stabilisers of the three partitions
$\{\{1,2\},\{3,4\}\}$, $\{\{1,3\},\{2,4\}\}$, $\{\{1,4\},\{2,3\}\}$ are dihedral of order $8$, so $n_2=3$ (it divides $3$ and is not $1$).

(c) In $\GL_2(\F_3)$ (order $48=2^4\cdot3$): $n_3=4$, the four subgroups $\mathrm U_L$ of \Cref{prop:groups:unipotent}(v), one for each line $L$ of $\F_3^2$; and
$n_2=3$, the three conjugates of $N_{ns}(\F_3)$ (\Cref{prop:groups:nns3}(iii)).

(d) In $\SL_2(\F_3)$ (order $24$): $n_2=1$, the Sylow $2$-subgroup being $Q_8$ (\Cref{prop:groups:sl23}(ii)), and $n_3=4$.
\end{example}

\section{Finite abelian groups}\label{sec:groups:abelian}

The paper's local invariants are orders of finite abelian groups and their $\ell$-parts: the geometric component groups $\Phig_q$, of the
shapes $\Z/n$, $\Z/2$, $\Z/3$, $(\Z/2)^2$, $\Z/4$ (\Cref{ch:reduction}), the Tamagawa numbers, and the torsion subgroup
$H^1(I_q,M)_{\mathrm{tor}}\simeq(\Z/\ell^{v_\ell(n)})^2$ in the proof of Lemma lem:tamfactor, whose Frobenius-fixed points are counted there.
This section proves the structure theorem and the counting rule $\#A[n]=\#A/nA$ used for that count, and shows that the multiplicative group of
a finite field is cyclic, which Lemma lem:h0 uses for $\F_\ell^\times$. Groups are written additively; $A[n]=\{a\in A:na=0\}$.

\begin{theorem}[Structure of finite abelian groups]\label{thm:groups:finiteabelian}
Every finite abelian group $A$ is isomorphic to $\bigoplus_{i=1}^s\Z/\ell_i^{e_i}$ for primes $\ell_i$ and integers $e_i\ge1$, and the multiset of prime
powers $\ell_i^{e_i}$ (the \emph{elementary divisors}) is determined by $A$. Equivalently, $A\simeq\Z/d_1\oplus\dots\oplus\Z/d_r$ with
$1<d_1\mid d_2\mid\dots\mid d_r$, and the \emph{invariant factors} $d_i$ are determined by $A$. For each prime $\ell$, the numbers $\#A[\ell^k]$, $k\ge1$,
determine $A[\ell^\infty]$ up to isomorphism: the number of $i$ with $\ell_i=\ell$ and $e_i\ge k$ is $\log_\ell\bigl(\#A[\ell^k]/\#A[\ell^{k-1}]\bigr)$.
\end{theorem}

\begin{proof}
By \Cref{prop:groups:lprimary}(iv), $A=\bigoplus_\ell A[\ell^\infty]$, so it suffices to treat a finite abelian $\ell$-group $A$.

\emph{Existence.} We show by induction on $\#A$: if $a\in A$ has maximal order $\ell^e$, then $A=\langle a\rangle\oplus B$ for some subgroup $B$; applying
this again to $B$ gives the decomposition. If $A=\langle a\rangle$, take $B=0$. Otherwise choose $b\notin\langle a\rangle$ of least possible order. Then $\ell b$ has smaller
order, so $\ell b\in\langle a\rangle$, say $\ell b=ta$. Since $\ell^{e-1}(ta)=\ell^eb=0$ (the order of $b$ is at most $\ell^e$), the element $ta$ is not a generator of
$\langle a\rangle$, so $\ell\mid t$, say $t=\ell t'$. Put $c=b-t'a$. Then $\ell c=0$ and $c\notin\langle a\rangle$, so $C=\langle c\rangle$ has order $\ell$ and
$C\cap\langle a\rangle=0$. In $\bar A=A/C$ the image $\bar a$ has order $\ell^e$ (if $\ell^{e-1}a\in C$, then $\ell^{e-1}a\in C\cap\langle a\rangle=0$), and this is the maximal
order in $\bar A$, as orders can only drop in a quotient. By induction $\bar A=\langle\bar a\rangle\oplus\bar B$; let $B$ be the preimage of $\bar B$, which contains $C$.
If $ma\in B$, then $m\bar a\in\langle\bar a\rangle\cap\bar B=0$, so $ma\in C\cap\langle a\rangle=0$; thus $\langle a\rangle\cap B=0$. For $x\in A$ write $\bar x=m\bar a+\bar y$ with $y\in B$; then
$x-ma-y\in C\subseteq B$, so $x\in\langle a\rangle+B$. Hence $A=\langle a\rangle\oplus B$.

\emph{Uniqueness.} If $A\simeq\bigoplus_i\Z/\ell^{e_i}$, then $A[\ell^k]\simeq\bigoplus_i(\Z/\ell^{e_i})[\ell^k]$, and $(\Z/\ell^e)[\ell^k]$ has order $\ell^{\min(k,e)}$
(\Cref{thm:groups:cyclic}(i)). So $\#A[\ell^k]=\ell^{\sum_i\min(k,e_i)}$, and $\#A[\ell^k]/\#A[\ell^{k-1}]=\ell^{\#\{i:e_i\ge k\}}$. The numbers
$\#\{i:e_i\ge k\}$, $k\ge1$, determine the multiset $\{e_i\}$.

\emph{Invariant factors.} Given the elementary divisors, list for each $\ell$ the exponents in decreasing order $e_{\ell,1}\ge e_{\ell,2}\ge\cdots$ (padded with
zeros), and put $d_r=\prod_\ell\ell^{e_{\ell,1}}$, $d_{r-1}=\prod_\ell\ell^{e_{\ell,2}}$, and so on. By \Cref{prop:groups:crtgroups}, $\Z/d_j\simeq\bigoplus_\ell\Z/\ell^{e_{\ell,r+1-j}}$, so
$A\simeq\bigoplus_j\Z/d_j$, with $d_1\mid\dots\mid d_r$. Conversely the elementary divisors of $\bigoplus_j\Z/d_j$ are the prime powers exactly dividing the
$d_j$, so the $d_j$ are determined by the elementary divisors, hence by $A$.
\end{proof}

\begin{corollary}[Cyclicity criteria]\label{cor:groups:cyclic}
\begin{enumerate}[label=(\roman*)]
\item For a finite abelian group $A$ the following are equivalent: $A$ is cyclic; $\#A[\ell]\le\ell$ for every prime $\ell$; $\#A[d]\le d$ for every $d\ge1$.
\item Every finite subgroup of the multiplicative group $K^\times$ of a field $K$ is cyclic. In particular $\F_p^\times$ is cyclic of order $p-1$, more
generally $\F_q^\times$ is cyclic of order $q-1$, the group $\mu_n(K)$ of $n$-th roots of unity in $K$ is cyclic, and the multiplicative group of the
field $\F_q[\sqrt\varepsilon]$ of \Cref{def:groups:cartan} is cyclic of order $q^2-1$.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) If $A\simeq\Z/n$, then $\#A[d]=\gcd(d,n)\le d$ by \Cref{thm:groups:cyclic}(ii). The third condition implies the second. If $\#A[\ell]\le\ell$ for all $\ell$, then
in the decomposition of \Cref{thm:groups:finiteabelian} each $A[\ell^\infty]\simeq\bigoplus_i\Z/\ell^{e_i}$ has $\#A[\ell]=\ell^{\#\{i\}}$, so it has at most one
summand; hence $A\simeq\bigoplus_\ell\Z/\ell^{e_\ell}$, which is cyclic by \Cref{prop:groups:crtgroups}. (ii) Let $A\le K^\times$ be finite. The elements of $A$ with
$x^d=1$ are roots of the polynomial $X^d-1$, which has at most $d$ roots in the field $K$; so $\#A[d]\le d$ for all $d$, and $A$ is cyclic by (i).
\end{proof}

\begin{proposition}[Counting torsion]\label{prop:groups:torsioncount}
Let $A$ be a finite abelian group and $n\ge1$. The sequence $0\to A[n]\to A\xrightarrow{\,n\,}A\to A/nA\to0$ is exact, and $\#A[n]=\#(A/nA)$. If
$A\simeq\bigoplus_i\Z/\ell^{e_i}$ is an $\ell$-group, then $\#A[\ell^k]=\#(A/\ell^kA)=\ell^{\sum_i\min(k,e_i)}$.
\end{proposition}

\begin{proof}
Exactness is the definition of $A[n]$ (the kernel of multiplication by $n$) and of $A/nA$ (its cokernel). By the first isomorphism theorem
$\#nA=\#A/\#A[n]$, so $\#(A/nA)=\#A/\#nA=\#A[n]$. The formula for $\#A[\ell^k]$ was proved in the proof of \Cref{thm:groups:finiteabelian}.
\end{proof}

\begin{paperlink}
In the proof of Lemma lem:tamfactor the Frobenius-fixed part of $(\Z/\ell^{v_\ell(n)})^2$ is described by two congruences, $(q-1)y\equiv0$ and
$c'y\equiv0\pmod{\ell^{v_\ell(n)}}$, for $y\in\Z/\ell^{v_\ell(n)}$. The solutions form $(\Z/\ell^{v})[\ell^{w}]$ with $v=v_\ell(n)$ and
$w=\min(v_\ell(q-1),v_\ell(c'))$, a subgroup of order $\ell^{\min(v,w)}=\ell^{\min(v_\ell(n),v_\ell(q-1),v_\ell(c'))}$ by
\Cref{thm:groups:cyclic}(i) and \Cref{prop:groups:torsioncount}: indeed $\gamma y\equiv0\pmod{\ell^v}$ holds iff $\ell^{\min(v,v_\ell(\gamma))}y\equiv0$, by
\Cref{thm:groups:cyclic}(ii). The lemma itself is proved in \Cref{ch:blochkato}; here only the count is at stake.
\end{paperlink}

\begin{example}\label{ex:groups:componentgroups}
The geometric component groups of an elliptic curve (\Cref{ch:reduction}) are $\Z/n$ (type $\kod{I}_n$), $0$ (types $\kod{II}$, $\kod{II}^*$), $\Z/2$
(types $\kod{III}$, $\kod{III}^*$), $\Z/3$ (types $\kod{IV}$, $\kod{IV}^*$), and $(\Z/2)^2$ or $\Z/4$ (types $\kod{I}_n^*$, according to the parity of $n$).
Their $\ell$-parts are, by \Cref{prop:groups:lprimary}, of order $\ell^{v_\ell(n)}$ for $\Z/n$; for odd $\ell$ only $\Z/n$ and $\Z/3$ (when $\ell=3$) can
contribute. The groups $(\Z/2)^2$ and $\Z/4$ have the same order but are not isomorphic, since $\#(\Z/2)^2[2]=4$ and $\#(\Z/4)[2]=2$. The paper's Section 1.4 and
Definition def:terms use only the numbers $v_\ell(\#\Phig_q)$.
\end{example}

\section[Matrix groups over a field]{The groups \texorpdfstring{$\GL_2$, $\SL_2$, $\PGL_2$, $\PSL_2$}{GL2, SL2, PGL2, PSL2}; Borel, unipotent and Cartan subgroups}\label{sec:groups:matrix}

The Galois representation $\bar\rho_{E,\ell}$ takes values in $\GL_2(\F_\ell)$, and the paper's arguments at $\ell=3$ are statements about its
subgroups. This section introduces the subgroups that occur: the Borel subgroups (the image when $E$ has a rational $\ell$-isogeny, the
Eisenstein case of Section sec:eis and Conjecture conj:eis), the unipotent elements (the image of inertia in Lemma lem:IV, and ``minus a
unipotent'' in Remark rem:types), the split and nonsplit Cartan subgroups and their normalisers (Proposition prop:ns, Section sec:anomaly,
Remark rem:ns), and the action on the projective line, which carries the four lines of $\F_3^2$ of Proposition prop:ns. Throughout, $F$ is a
field and $\F_q$ a finite field with $q$ elements (convention (C4)).

\begin{definition}[Linear groups]\label{def:groups:gl2}
$\GL_2(F)$ is the group of invertible $2\times2$ matrices over $F$; $\SL_2(F)=\ker(\det\colon\GL_2(F)\to F^\times)$; $\PGL_2(F)=\GL_2(F)/F^\times I$; and
$\PSL_2(F)=\SL_2(F)/\{\pm I\}$.
\end{definition}

\begin{proposition}\label{prop:groups:gl2}
\begin{enumerate}[label=(\roman*)]
\item $Z(\GL_2(F))=F^\times I$ and $Z(\SL_2(F))=\{\pm I\}$.
\item $\det$ is surjective, $d\mapsto\mathrm{diag}(1,d)$ is a homomorphic section, and $\GL_2(F)\simeq\SL_2(F)\rtimes F^\times$.
\item $\SL_2(F)$ is generated by the elementary matrices $\sm1b01$ and $\sm10c1$, $b,c\in F$.
\item If $\operatorname{char}F\ne2$, the only element of order $2$ of $\SL_2(F)$ is $-I$.
\item The image of $\SL_2(F)$ in $\PGL_2(F)$ is isomorphic to $\PSL_2(F)$, and its index is $[F^\times:(F^\times)^2]$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Let $g=\sm abcd$ commute with $e=\sm1101$: $ge=\sm a{a+b}c{c+d}$ and $eg=\sm{a+c}{b+d}cd$, so $c=0$ and $a=d$. Commuting with $\sm1011$ gives
$b=0$ in the same way. So $g$ is scalar; conversely scalars are central. Both elementary matrices lie in $\SL_2(F)$, so the same computation gives
$Z(\SL_2(F))=\{aI:a^2=1\}=\{\pm I\}$. (ii) $\det\mathrm{diag}(1,d)=d$, and $d\mapsto\mathrm{diag}(1,d)$ is a homomorphism; apply \Cref{prop:groups:semidirect}(ii).
(iii) Let $g=\sm abcd\in\SL_2(F)$. If $c=0$, multiplying on the left by $\sm1011$ replaces $c$ by $a\ne0$. So assume $c\ne0$. Left multiplication by
$\sm1x01$ with $x=(1-a)/c$ makes the top left entry $1$; then left multiplication by $\sm10{-c}1$ clears the bottom left entry, leaving $\sm1{b'}0{d'}$
with $d'=1$ (determinant $1$); and left multiplication by $\sm1{-b'}01$ gives $I$. So $g$ is a product of elementary matrices. (iv) If $g^2=I$, the
minimal polynomial of $g$ divides $X^2-1=(X-1)(X+1)$, which has distinct roots as $1\ne-1$; so $g$ is diagonalisable with eigenvalues in $\{\pm1\}$, and
$\det g=1$ forces them to be equal, so $g=\pm I$. (v) $\SL_2(F)\cap F^\times I=\{\pm I\}$, so the image is $\SL_2(F)F^\times/F^\times\simeq\SL_2(F)/\{\pm I\}$ by
\Cref{thm:groups:isothms}(ii). The index is $[\GL_2(F):\SL_2(F)F^\times I]$, and $\det$ induces an isomorphism $\GL_2(F)/\SL_2(F)F^\times I\simeq F^\times/(F^\times)^2$,
as $\det(\lambda I)=\lambda^2$.
\end{proof}

The next lemma is the linear algebra behind every conjugacy statement in this chapter.

\begin{lemma}[Companion form and centralisers]\label{lem:groups:companion}
Let $g\in\GL_2(F)$ be non-scalar. Then there is a vector $v$ such that $(v,gv)$ is a basis, and in this basis $g$ has matrix $\sm0{-\det g}1{\tr g}$. Hence two
non-scalar matrices are conjugate in $\GL_2(F)$ if and only if they have the same characteristic polynomial. The centraliser of $g$ in $\GL_2(F)$ is
$F[g]^\times$, the unit group of the two-dimensional commutative algebra $F[g]=F+Fg\simeq F[X]/(X^2-\tr(g)X+\det g)$. Over $F=\F_q$:
$\#C(g)=q^2-1$ if the characteristic polynomial is irreducible, $(q-1)^2$ if it has two distinct roots, and $q(q-1)$ if it has a double root.
\end{lemma}

\begin{proof}
If every nonzero vector were an eigenvector, then $ge_1=\alpha e_1$, $ge_2=\beta e_2$ and $g(e_1+e_2)=\gamma(e_1+e_2)$ would give $\alpha=\gamma=\beta$, and $g$ would
be scalar. So some $v$ is not an eigenvector, and $(v,gv)$ is a basis. By the Cayley-Hamilton theorem $g^2v=\tr(g)\,gv-\det(g)\,v$, which gives the
matrix. Two non-scalar matrices with the same characteristic polynomial are therefore both conjugate to the same companion matrix. If $h$
commutes with $g$, write $hv=\alpha v+\beta gv$; then $h(gv)=g(hv)=(\alpha+\beta g)gv$, so $h$ and $\alpha+\beta g$ agree on a basis: $h\in F[g]$. Conversely $F[g]$
commutes with $g$. As $g$ is not scalar, $I$ and $g$ are linearly independent and $F[g]\simeq F[X]/(\chi_g)$ with $\chi_g$ the characteristic polynomial
(the minimal polynomial has degree $2$). Over $\F_q$ the unit group of $\F_q[X]/(\chi)$ has order $q^2-1$ if $\chi$ is irreducible (a field with $q^2$
elements), $(q-1)^2$ if $\chi=(X-\lambda)(X-\mu)$ with $\lambda\ne\mu$ (the ring is $\F_q\times\F_q$ by the Chinese remainder theorem), and $q^2-q$ if $\chi=(X-\lambda)^2$
(the non-units form the ideal generated by $X-\lambda$, with $q$ elements).
\end{proof}

\begin{proposition}[Orders]\label{prop:groups:orders}
$\#\GL_2(\F_q)=(q^2-1)(q^2-q)=q(q-1)^2(q+1)$, $\#\SL_2(\F_q)=\#\PGL_2(\F_q)=q(q^2-1)$, and $\#\PSL_2(\F_q)=q(q^2-1)/\gcd(2,q-1)$. More generally
$\#\GL_n(\F_q)=\prod_{i=0}^{n-1}(q^n-q^i)$. For small $q$:
\begin{center}
\begin{tabular}{ccccc}
\toprule
$q$ & $\#\GL_2(\F_q)$ & $\#\SL_2(\F_q)$ & $\#\PGL_2(\F_q)$ & $\#\PSL_2(\F_q)$\\
\midrule
$3$ & $48$ & $24$ & $24$ & $12$\\
$5$ & $480$ & $120$ & $120$ & $60$\\
$7$ & $2016$ & $336$ & $336$ & $168$\\
\bottomrule
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}
An invertible matrix is an ordered basis of columns: the first column is any of the $q^2-1$ nonzero vectors, the second any of the $q^2-q$ vectors
outside the line of the first; for $\GL_n$ the $i$-th column avoids the $q^{i-1}$ vectors of the span of the previous ones. $\det$ is surjective onto
$\F_q^\times$ (\Cref{prop:groups:gl2}(ii)), so $\#\SL_2=\#\GL_2/(q-1)$; $\#F^\times I=q-1$ gives $\#\PGL_2$; and $\#\{\pm I\}=\gcd(2,q-1)$ gives $\#\PSL_2$.
\end{proof}

\begin{proposition}[The projective line]\label{prop:groups:projline}
Let $\PP^1(F)$ be the set of lines of $F^2$ (convention (C1)). Then $\#\PP^1(\F_q)=q+1$, and:
\begin{enumerate}[label=(\roman*)]
\item $\GL_2(F)$ acts on $\PP^1(F)$ by $g[v]=[gv]$; in the coordinate $x=[x:1]$, $\infty=[1:0]$, the matrix $\sm abcd$ acts by $x\mapsto(ax+b)/(cx+d)$,
with the usual conventions $a/0=\infty$ for $a\ne0$ and $\infty\mapsto a/c$.
\item The kernel of the action is $F^\times I$, so $\PGL_2(F)$ acts faithfully.
\item $\PGL_2(F)$ acts sharply $3$-transitively: for two triples of distinct lines there is exactly one element of $\PGL_2(F)$ carrying the first to
the second.
\item The stabiliser of $\infty=[e_1]$ is the upper triangular group $\mathrm B(F)$; so $\PP^1(F)\simeq\GL_2(F)/\mathrm B(F)$ and $[\GL_2(\F_q):\mathrm B(\F_q)]=q+1$.
\item $\SL_2(F)$ acts $2$-transitively, in particular transitively, on $\PP^1(F)$.
\end{enumerate}
\end{proposition}

\begin{proof}
There are $q^2-1$ nonzero vectors, $q-1$ on each line, so $q+1$ lines. (i) $g(\lambda v)=\lambda gv$, so the action is well defined. $g\binom x1=\binom{ax+b}{cx+d}$,
which is the line $(ax+b)/(cx+d)$ if $cx+d\ne0$ and $\infty$ otherwise; and $g\binom10=\binom ac$. (ii) An element acting trivially has every nonzero vector
as an eigenvector, so it is scalar by the first step of the proof of \Cref{lem:groups:companion}. (iii) Let $L_1,L_2,L_3$ be distinct lines, $L_i=[v_i]$. Then
$v_1,v_2$ is a basis and $v_3=\alpha v_1+\beta v_2$ with $\alpha,\beta\ne0$ (as $L_3\ne L_1,L_2$). The matrix $g$ with columns $\alpha v_1$, $\beta v_2$ is invertible and
maps $[e_1],[e_2],[e_1+e_2]$, that is $\infty,0,1$, to $L_1,L_2,L_3$. If $h$ fixes $\infty$, $0$ and $1$, then $he_1=\alpha e_1$, $he_2=\beta e_2$ and
$h(e_1+e_2)=\gamma(e_1+e_2)$, so $\alpha=\beta=\gamma$ and $h$ is scalar. So for each triple there is exactly one element of $\PGL_2(F)$ carrying $(\infty,0,1)$ to it, and
composing gives (iii). (iv) $g[e_1]=[e_1]$ iff the first column $\binom ac$ is a multiple of $e_1$, iff $c=0$. Then apply \Cref{thm:groups:orbitstab} and (iii)
(which gives transitivity). (v) Given two ordered pairs of distinct lines, it suffices to move $(\infty,0)$ to any pair $([v_1],[v_2])$: the matrix with columns
$v_1$ and $\lambda v_2$, where $\lambda=\det(v_1,v_2)^{-1}$, lies in $\SL_2(F)$ and does this.
\end{proof}

\begin{definition}[Borel subgroups]\label{def:groups:borel}
A \emph{Borel subgroup} of $\GL_2(F)$ is the stabiliser $\mathrm B_L$ of a line $L\in\PP^1(F)$. We write $\mathrm B=\mathrm B_{[e_1]}=\{\sm ab0d\}$, the upper triangular
group; $\mathrm U=\{\sm1b01:b\in F\}$, its \emph{unipotent radical}; and $\mathrm T=\{\mathrm{diag}(a,d)\}$, the diagonal torus (which is the split Cartan subgroup $C_s$ of
\Cref{def:groups:cartan}).
\end{definition}

\begin{proposition}\label{prop:groups:borel}
\begin{enumerate}[label=(\roman*)]
\item $\mathrm U\simeq(F,+)$ is normal in $\mathrm B$, and $\mathrm B=\mathrm U\rtimes\mathrm T$, with $\mathrm T$ acting on $\mathrm U$ by
\[\mathrm{diag}(a,d)\sm1b01\mathrm{diag}(a,d)^{-1}=\sm1{ab/d}01;\]
and $\#\mathrm B(\F_q)=q(q-1)^2$.
\item The Borel subgroups are all conjugate: $g\mathrm B_Lg^{-1}=\mathrm B_{gL}$. A Borel subgroup stabilises exactly one line, so $L\mapsto\mathrm B_L$ is a bijection from
$\PP^1(F)$ to the set of Borel subgroups, and $N_{\GL_2(F)}(\mathrm B_L)=\mathrm B_L$.
\item A subgroup of $\GL_2(F)$ stabilises a line of $F^2$ if and only if it is contained in a Borel subgroup.
\item For $q=3$: $\#\mathrm B(\F_3)=12$, and $\mathrm B(\F_3)=\{\pm I\}\times\langle u,\sigma\rangle\simeq C_2\times S_3\simeq D_6$, with $u=\sm1101$ and $\sigma=\mathrm{diag}(1,-1)$. Its image in
$\PGL_2(\F_3)$ is the stabiliser of $\infty$, isomorphic to $S_3$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The map $\sm ab0d\mapsto\mathrm{diag}(a,d)$ is a homomorphism $\mathrm B\to\mathrm T$ (the diagonal of a product of upper triangular matrices is the product of the
diagonals) with kernel $\mathrm U$, and it restricts to the identity on $\mathrm T$; so \Cref{prop:groups:semidirect}(ii) applies. The conjugation formula is a direct
computation, and $\sm1b01\sm1{b'}01=\sm1{b+b'}01$. The order is $q\cdot(q-1)^2$. (ii) $g\mathrm B_Lg^{-1}$ stabilises $gL$, and conversely; the action on lines is
transitive. The element $\sm1101$ of $\mathrm B$ acts by $x\mapsto x+1$, which fixes $\infty$ and no point of $F$; so $\mathrm B$ stabilises only $\infty$, and by conjugation $\mathrm B_L$ stabilises only $L$. If $g$ normalises $\mathrm B_L$, then $\mathrm B_L=g\mathrm B_Lg^{-1}=\mathrm B_{gL}$, so $gL=L$ and
$g\in\mathrm B_L$. (iii) is the definition. (iv) $\#\mathrm B(\F_3)=3\cdot4$. Since $\sigma u\sigma^{-1}=\sm1{-1}01=u^{-1}$, the subgroup $S=\langle u,\sigma\rangle=\{u^m\sigma^e\}$ has order $6$ and is
isomorphic to $D_3\simeq S_3$ by \Cref{prop:groups:dihedral}(i),(iv). It has index $2$ in $\mathrm B(\F_3)$, hence is normal; $\{\pm I\}$ is central; and $-I\notin S$
(the elements $u^m$ have trace $2$, the elements $u^m\sigma$ determinant $-1$, while $-I$ has trace $-2=1$ and determinant $1$). So
$\mathrm B(\F_3)=\{\pm I\}\times S$ by \Cref{prop:groups:directproduct}, and $C_2\times S_3\simeq D_6$ by \Cref{prop:groups:dihedral}(iv). The image in $\PGL_2(\F_3)$ is
$\mathrm B(\F_3)/\{\pm I\}\simeq S$, of order $6$, contained in the stabiliser of $\infty$, which has order $24/4=6$ by \Cref{thm:groups:orbitstab}.
\end{proof}

\begin{definition}[Unipotent elements]\label{def:groups:unipotent}
An element $u\in\GL_2(F)$ is \emph{unipotent} if $(u-1)^2=0$; equivalently, its characteristic polynomial is $(X-1)^2$ (if $(u-1)^2=0$, the only
eigenvalue of $u$ over $\overline F$ is $1$; conversely apply the Cayley-Hamilton theorem). For a line $L$, $\mathrm U_L$ denotes the group of unipotent elements
fixing $L$ pointwise; $\mathrm U_{[e_1]}=\mathrm U$.
\end{definition}

\begin{proposition}\label{prop:groups:unipotent}
\begin{enumerate}[label=(\roman*)]
\item A unipotent $u\ne1$ is conjugate to $\sm1101$. Its fixed vectors form exactly one line, $\ker(u-1)=\im(u-1)$, and $u$ stabilises no other line
of $F^2$.
\item If $\operatorname{char}F=p>0$, a unipotent $u\ne1$ has order $p$; conversely every element of $p$-power order of $\GL_2(F)$ is unipotent.
\item The unipotent elements of $\mathrm B$ are exactly the elements of $\mathrm U$. For each line $L$, $\mathrm U_L=g\mathrm Ug^{-1}$ for any $g$ with $g[e_1]=L$; it is the unipotent
radical of $\mathrm B_L$ and is isomorphic to $(F,+)$. The stabiliser of a nonzero vector $v$ in $\SL_2(F)$ is $\mathrm U_{[v]}$.
\item (Minus a unipotent.) If $u\ne1$ is unipotent, $-u$ has characteristic polynomial $(X+1)^2$; if $\operatorname{char}F\ne2$, it has no nonzero fixed vector, it
stabilises exactly one line (that of $u$), and in characteristic $p$ odd it has order $2p$.
\item In $\GL_2(\F_3)$ the $8$ elements of order $3$ are the nontrivial unipotent elements and form one conjugacy class. They lie in the four
subgroups $\mathrm U_L$ of order $3$, one for each line $L$, which are the Sylow $3$-subgroups; $\mathrm U_L$ fixes exactly the line $L$. The $8$ elements
of order $6$ are the elements $-u$ with $u\ne1$ unipotent. The $8$ elements of order $3$ generate $\SL_2(\F_3)$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Let $N=u-1\ne0$. As $N^2=0$, $\im N\subseteq\ker N$, and $N$ has rank $1$ (rank $2$ would make $N^2$ invertible); so $\ker N=\im N$ is a line $L$. Choose
$v\notin L$ and put $w=Nv$, which spans $L$. In the basis $(w,v)$, $uw=w$ and $uv=v+w$, so $u$ has matrix $\sm1101$. The fixed vectors are $\ker N=L$. A line
stabilised by $u$ is spanned by an eigenvector, whose eigenvalue is $1$, so it is $L$. (ii) Write $u=1+N$ with $N^2=0$; then $u^m=1+mN$, which is $1$ iff
$p\mid m$. Conversely if $x^{p^k}=1$, then $(x-1)^{p^k}=x^{p^k}-1=0$ (the binomial coefficients $\binom{p^k}i$, $0<i<p^k$, vanish in characteristic $p$), so $x-1$
is nilpotent; a nilpotent $2\times2$ matrix has characteristic polynomial $X^2$ and squares to $0$ by Cayley-Hamilton. (iii) An upper triangular matrix
with characteristic polynomial $(X-1)^2$ has diagonal entries $1$. A unipotent element fixing $L=g[e_1]$ pointwise is $gu'g^{-1}$ with $u'$ unipotent fixing
$e_1$, hence in $\mathrm B$, hence in $\mathrm U$; conversely $\mathrm U$ fixes $e_1$. So $\mathrm U_L=g\mathrm Ug^{-1}$, the kernel of $\mathrm B_L\to\mathrm T$ transported by $g$. If $g'\in\SL_2(F)$
fixes $v\ne0$, complete $v$ to a basis $(v,v')$; in it $g'=\sm1b0d$ with $d=\det g'=1$, so $g'$ is unipotent and fixes $v$; conversely elements of $\mathrm U_{[v]}$
fix $v$ and have determinant $1$. (iv) $\det(X+u)=(X+1)^2$ follows from $\det(X-u)=(X-1)^2$. If $-uv=v$ then $uv=-v$, but $u$ has only the eigenvalue
$1\ne-1$. The lines stabilised by $-u$ are those stabilised by $u$. Finally $(-u)^m=(-1)^mu^m=1$ iff $m$ is even and $p\mid m$, so the order is $2p$ for
odd $p$. (v) By (ii) the elements of order $3$ are the nontrivial unipotents; by (i) and \Cref{lem:groups:companion} they form one class, of size
$48/\#C(u)=48/(3\cdot2)=8$. Each lies in exactly one $\mathrm U_L$ (its fixed line), and $\#\mathrm U_L=3$; so the four groups $\mathrm U_L$ contain $4\cdot2=8$ elements of order $3$,
and they are the subgroups of order $3$, that is, the Sylow $3$-subgroups ($48=16\cdot3$). If $x$ has order $6$, then $x^3$ has order $2$ and $x^{-2}$ has order
$3$, so $x^{-2}=u'$ is a nontrivial unipotent, and $x^3$ is an involution commuting with $u'$, hence in $C(u')=\F_3[u']^\times=\{\pm u'^m\}$; its only involution
is $-I$ (the elements $-u'^m$ with $m\ne0$ have order $6$ by (iv)). So $x=x^3x^{-2}=-u'$. Conversely the $-u$ have order $6$ by (iv); there are $8$.
Finally let $S$ be the subgroup generated by the $8$ elements of order $3$. It lies in $\SL_2(\F_3)$ (unipotents have determinant $1$), so $\#S$ divides $24$, and
$\#S\ge9$; hence $\#S\in\{12,24\}$. If $\#S=12$, then $S$ has index $2$ in $\SL_2(\F_3)$, hence contains the square of every element of $\SL_2(\F_3)$
(\Cref{lem:groups:index2}: the quotient has order $2$); in particular $-I=J^2$ with $J=\sm0{-1}10$, and then also the $8$ elements $-u$. So $\#S\ge1+8+1+8=18$, a
contradiction. Hence $S=\SL_2(\F_3)$.
\end{proof}

\begin{lemma}[Commutators in a Borel subgroup]\label{lem:groups:borelcommutator}
Let $K$ be a field and $H\le\mathrm B(K)$. Then $[H,H]\le\mathrm U(K)$. If $\operatorname{char}K=p>0$, then $\mathrm U(K)$ is an elementary abelian $p$-group (every nontrivial element
has order $p$), and:
\begin{enumerate}[label=(\roman*)]
\item a finite subgroup of $\mathrm B(K)$ of order prime to $p$ meets $\mathrm U(K)$ trivially and is abelian;
\item a finite subgroup of $\GL_2(K)$ whose commutator subgroup is not a $p$-group stabilises no line of $K^2$.
\end{enumerate}
\end{lemma}

\begin{proof}
The homomorphism $\mathrm B(K)\to\mathrm T(K)\simeq(K^\times)^2$ of \Cref{prop:groups:borel}(i) has abelian target and kernel $\mathrm U(K)$, so it kills all commutators:
$[H,H]\le[\mathrm B,\mathrm B]\le\mathrm U(K)$. In characteristic $p$, $\mathrm U(K)\simeq(K,+)$ and $p\cdot b=0$. (i) $H\cap\mathrm U(K)$ is a $p$-group (each element has order $1$ or $p$)
whose order divides $\#H$, so it is trivial, and $[H,H]\le H\cap\mathrm U(K)=1$. (ii) If $H$ stabilises a line of $K^2$, then $gHg^{-1}\le\mathrm B(K)$ for some $g\in\GL_2(K)$
(\Cref{prop:groups:borel}(ii),(iii)), so $g[H,H]g^{-1}\le\mathrm U(K)$, and $[H,H]$ is a finite subgroup of a conjugate of $\mathrm U(K)$, hence a $p$-group.
\end{proof}

\begin{theorem}[$\PGL_2$ and $\PSL_2$ over small fields]\label{thm:groups:smallpgl}
\begin{enumerate}[label=(\roman*)]
\item The action on the four lines of $\F_3^2$ induces isomorphisms $\PGL_2(\F_3)\simeq S_4$ and $\PSL_2(\F_3)\simeq A_4$.
\item For $g\in\GL_2(\F_3)$, $\det g\in\F_3^\times=\{\pm1\}$ equals the sign of the permutation of the four lines induced by $g$.
\item $\PGL_2(\F_5)\simeq S_5$ and $\PSL_2(\F_5)\simeq A_5$.
\item $\PSL_2(\F_q)$ is simple for $q\ge4$. In particular $\PSL_2(\F_7)$ is simple of order $168$; it is isomorphic to $\GL_3(\F_2)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) By \Cref{prop:groups:projline}(ii) the action gives an injective homomorphism $\PGL_2(\F_3)\to S_4$, and both groups have order $24$. The group $\SL_2(\F_3)$
is generated by its elements of order $3$ (\Cref{prop:groups:unipotent}(v)), whose images have order $3$ in $S_4$, hence are $3$-cycles, hence even; so the image
of $\SL_2(\F_3)$, which is $\PSL_2(\F_3)$ by \Cref{prop:groups:gl2}(v), lies in $A_4$, and it has order $12$. (ii) Both $\det$ and $g\mapsto\sgn(\text{permutation})$ are
homomorphisms $\GL_2(\F_3)\to\{\pm1\}$. The kernel of $\det$ is $\SL_2(\F_3)$, which lies in the kernel of the sign by (i). The sign is not trivial: $\sigma=\mathrm{diag}(1,-1)$
acts by $x\mapsto-x$, fixing $0$ and $\infty$ and swapping $1$ and $-1$, a transposition. So both kernels have index $2$ and contain $\SL_2(\F_3)$, hence are equal to it,
and two homomorphisms onto $\{\pm1\}$ with the same kernel are equal.
\end{proof}

We do not prove (iv); see \cite[Chapter XIII, Section 8]{Lang} for the simplicity of $\PSL_2(F)$ for every field with more than three elements
(the proof uses the $2$-transitivity of \Cref{prop:groups:projline}(v), the generation by the conjugates of $\mathrm U$ of \Cref{prop:groups:gl2}(iii), and Iwasawa's
criterion), and \cite[Section 6.2]{DummitFoote} for the uniqueness of the simple group of order $168$, which identifies $\PSL_2(\F_7)$ with $\GL_3(\F_2)$.

\begin{proof}[Sketch of proof of (iii)]
Granting the simplicity of $\PSL_2(\F_5)$ (order $60$): the subgroup of $\GL_2(\F_5)$ generated by $N_s(\F_5)$ (\Cref{def:groups:cartan}) and $\sm111{-1}$ has order
$96$ and contains the scalars (this is a finite check, done by machine; note $\sm111{-1}^2=2I$), so its image $K$ in $\PGL_2(\F_5)$ has order $24$ and index $5$. The
action of $\PGL_2(\F_5)$ on the five cosets of $K$ gives a homomorphism to $S_5$ whose kernel $M$ is normal and contained in $K$. Then $M\cap\PSL_2(\F_5)$ is normal in
the simple group $\PSL_2(\F_5)$ and has order at most $24<60$, so it is trivial, and $\#M\le2$; a normal subgroup of order $2$ would be central, but the centre of
$\PGL_2(\F_5)$ is trivial (a central element $z$ maps each point $x$ to a point fixed by the stabiliser of $x$, which is transitive on the other five points by
\Cref{prop:groups:projline}(iii); so $zx=x$ for all $x$, and $z=1$ by \Cref{prop:groups:projline}(ii)). So $\PGL_2(\F_5)\to S_5$ is injective, hence bijective, and it maps the index-$2$ subgroup $\PSL_2(\F_5)$ onto the unique index-$2$
subgroup $A_5$ of $S_5$. See \cite[Chapter II, Section 8]{Huppert67} for a full treatment of the small $\PSL_2(\F_q)$ and their exceptional isomorphisms.
\end{proof}

\begin{definition}[Cartan subgroups and their normalisers]\label{def:groups:cartan}
Let $q$ be odd and fix a non-square $\varepsilon\in\F_q^\times$ (for $q=3$ we take $\varepsilon=-1$; for $q=5$, $\varepsilon=2$; for $q=7$, $\varepsilon=3$). Put
$w=\sm0110$ and $\sigma=\mathrm{diag}(1,-1)$.
\begin{enumerate}[label=(\roman*)]
\item The \emph{split Cartan subgroup} is $C_s(\F_q)=\{\mathrm{diag}(a,d):a,d\in\F_q^\times\}\simeq(\F_q^\times)^2$, and $N_s(\F_q)=C_s(\F_q)\cup wC_s(\F_q)$.
\item Let $\F_q[\sqrt\varepsilon]=\{a+b\sqrt\varepsilon:a,b\in\F_q\}$, with $(\sqrt\varepsilon)^2=\varepsilon$, that is, $\F_q[X]/(X^2-\varepsilon)$. The \emph{nonsplit Cartan subgroup} is
\[
C_{ns}(\F_q)=\Bigl\{\Mtwo a{\varepsilon b}ba:(a,b)\ne(0,0)\Bigr\},
\]
the matrices of multiplication by $a+b\sqrt\varepsilon$ on $\F_q[\sqrt\varepsilon]$ in the basis $(1,\sqrt\varepsilon)$, and $N_{ns}(\F_q)=C_{ns}(\F_q)\cup C_{ns}(\F_q)\sigma$.
\end{enumerate}
A \emph{Cartan subgroup} of $\GL_2(\F_q)$ is a conjugate of $C_s(\F_q)$ (split) or of $C_{ns}(\F_q)$ (nonsplit). We often drop $(\F_q)$ from the notation.
\end{definition}

\begin{proposition}\label{prop:groups:cartan}
Let $q$ be odd.
\begin{enumerate}[label=(\roman*)]
\item $N_s=N_{\GL_2(\F_q)}(C_s)$, and it is the stabiliser of the set of two lines $\{[e_1],[e_2]\}$; $C_s$ is the subgroup fixing each of the two lines, and
$N_s=C_s\rtimes\langle w\rangle$.
\item $\F_q[\sqrt\varepsilon]$ is a field with $q^2$ elements, and $c\mapsto m_c$ (multiplication by $c$) is an isomorphism $\F_q[\sqrt\varepsilon]^\times\to C_{ns}$, with
$\det m_c=a^2-\varepsilon b^2$ (the norm $N(c)=c\bar c$, where $\bar c=a-b\sqrt\varepsilon$) and $\tr m_c=2a$. So $C_{ns}$ is cyclic of order $q^2-1$.
\item The conjugation $c\mapsto\bar c$ is the field automorphism $c\mapsto c^q$, its matrix is $\sigma$, and $\sigma m_c\sigma^{-1}=m_{\bar c}=m_c^q$. Moreover
$N_{ns}=N_{\GL_2(\F_q)}(C_{ns})=C_{ns}\rtimes\langle\sigma\rangle$.
\item All split Cartan subgroups are conjugate, and all nonsplit ones are conjugate. Intrinsically, a Cartan subgroup is the unit group $A^\times$ of a
two-dimensional commutative subalgebra $A\subset M_2(\F_q)$ that is isomorphic to $\F_q\times\F_q$ (split) or is a field (nonsplit). Each Cartan subgroup has
index $2$ in its normaliser.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $w\,\mathrm{diag}(a,d)\,w^{-1}=\mathrm{diag}(d,a)$, so $w$ normalises $C_s$, and $N_s$ is a group containing $C_s$ with index $2$ ($w\notin C_s$). An element fixing both $[e_1]$ and
$[e_2]$ is diagonal; an element swapping them is $w$ times a diagonal matrix. So $N_s$ is the stabiliser of $\{[e_1],[e_2]\}$. Conversely let $g$ normalise $C_s$. As
$q\ge3$, there is $d\ne1$ in $\F_q^\times$, and $\mathrm{diag}(1,d)$ has distinct eigenvalues, so the lines stabilised by every element of $C_s$ are exactly $[e_1]$ and $[e_2]$.
Since $gC_sg^{-1}=C_s$, $g$ maps this set of two lines to itself, so $g\in N_s$. The semidirect product structure follows from \Cref{prop:groups:semidirect}(i).

(ii) $\F_q[\sqrt\varepsilon]$ is a commutative ring with $q^2$ elements. For $(a,b)\ne(0,0)$, $a^2-\varepsilon b^2\ne0$: if $b\ne0$ this would make $\varepsilon=(a/b)^2$ a square, and if $b=0$
then $a\ne0$. So $(a+b\sqrt\varepsilon)^{-1}=(a-b\sqrt\varepsilon)/(a^2-\varepsilon b^2)$, and it is a field. Multiplication by $c=a+b\sqrt\varepsilon$ sends $1\mapsto a+b\sqrt\varepsilon$ and
$\sqrt\varepsilon\mapsto\varepsilon b+a\sqrt\varepsilon$, which gives the matrix $\sm a{\varepsilon b}ba$, with the stated determinant and trace; $c\mapsto m_c$ is an injective homomorphism.
Cyclicity is \Cref{cor:groups:cyclic}(ii).

(iii) $\bar c$ is a ring automorphism (it is the substitution $\sqrt\varepsilon\mapsto-\sqrt\varepsilon$, and $-\sqrt\varepsilon$ is also a root of $X^2-\varepsilon$), with matrix $\sigma$ in the basis
$(1,\sqrt\varepsilon)$. In characteristic $p$, $(x+y)^p=x^p+y^p$, so $c\mapsto c^q$ is a ring endomorphism; it fixes $\F_q$ (as $a^{q-1}=1$ for $a\in\F_q^\times$, by Lagrange) and sends
$\sqrt\varepsilon$ to $\varepsilon^{(q-1)/2}\sqrt\varepsilon$. Now $\varepsilon^{(q-1)/2}=-1$ (Euler's criterion): in the cyclic group $\F_q^\times=\langle t\rangle$ (\Cref{cor:groups:cyclic}) of even order $q-1$, the squares are
the even powers of $t$, since $q-1$ is even; so $\varepsilon=t^m$ with $m$ odd, and $\varepsilon^{(q-1)/2}=t^{m(q-1)/2}\ne1$, as $q-1\nmid m(q-1)/2$. Its square is $1$, and the only
roots of $X^2-1$ in a field are $\pm1$; so it is $-1$. Hence $c^q=\bar c$. Then $\sigma m_c\sigma^{-1}$
is the matrix of $x\mapsto\overline{c\,\bar x}=\bar cx$, that is $m_{\bar c}$. So $\sigma$ normalises $C_{ns}$; $\sigma\notin C_{ns}$ (it is not of the form $\sm a{\varepsilon b}ba$, as
$1\ne-1$), and $N_{ns}$ is a group in which $C_{ns}$ has index $2$. Conversely let $g$ normalise $C_{ns}$. The $\F_q$-span of $C_{ns}$ is the image $A$ of the field
$\F_q[\sqrt\varepsilon]$, and $x\mapsto gxg^{-1}$ is a ring automorphism of $A$ fixing the scalars; it sends $m_{\sqrt\varepsilon}$ to a square root of $\varepsilon I$ in $A$, that is to
$\pm m_{\sqrt\varepsilon}$ (a polynomial of degree $2$ has at most two roots in a field). If it fixes $m_{\sqrt\varepsilon}$, then $g$ commutes with the non-scalar matrix
$m_{\sqrt\varepsilon}$, so $g\in\F_q[m_{\sqrt\varepsilon}]^\times=C_{ns}$ by \Cref{lem:groups:companion}. Otherwise $g\sigma^{-1}$ commutes with $m_{\sqrt\varepsilon}$, so $g\in C_{ns}\sigma$.

(iv) Let $A\subset M_2(\F_q)$ be a two-dimensional commutative subalgebra (containing $I$). If $A\simeq\F_q\times\F_q$, its idempotent $e$ corresponding to $(1,0)$
satisfies $e^2=e\ne0,1$, so $e$ is the projection onto a line $L_1$ along a line $L_2$ ($F^2=\im e\oplus\ker e$, both of dimension $1$), and
$A=\{\alpha e+\beta(1-e)\}$ is the set of matrices acting by scalars on $L_1$ and on $L_2$; in a basis $v_1\in L_1$, $v_2\in L_2$, $A^\times$ becomes $C_s$. If $A$ is a field
(with $q^2$ elements), then $\F_q^2$ is an $A$-module, and for $v\ne0$ the map $A\to\F_q^2$, $x\mapsto xv$, is injective ($A$ is a field) and hence bijective by counting.
The element $\varepsilon\in\F_q^\times\subset A^\times$ is a square in $A$: $A^\times$ is cyclic of order $q^2-1$, its subgroup $\F_q^\times$ of order $q-1$ consists of the
$(q+1)$-th powers (\Cref{thm:groups:cyclic}(i)), and $q+1$ is even. Choose $\delta\in A$ with $\delta^2=\varepsilon I$; then in the basis $(v,\delta v)$ the elements $a+b\delta$ of $A$
have matrices $\sm a{\varepsilon b}ba$, so $A^\times$ becomes $C_{ns}$. Conversely $C_s$ and $C_{ns}$ are of this form, with $A$ the diagonal matrices and the image of
$\F_q[\sqrt\varepsilon]$. The index statement is (i) and (iii) transported by conjugation.
\end{proof}

\begin{proposition}[Facts about Cartan subgroups]\label{prop:groups:cartanfacts}
Let $q$ be odd.
\begin{enumerate}[label=(\roman*)]
\item $\#C_s=(q-1)^2$, $\#N_s=2(q-1)^2$, $\#C_{ns}=q^2-1$, $\#N_{ns}=2(q^2-1)$. For $q=3,5,7$ these are $4,16,36$; $8,32,72$; $8,24,48$; $16,48,96$, and
$\#\mathrm B=12,80,252$.
\item Every element of $N_s\setminus C_s$ and of $N_{ns}\setminus C_{ns}$ has trace $0$. In $N_s$, $\sm0bc0^2=bc\,I$ and $\det\sm0bc0=-bc$. In $N_{ns}$,
$(m_c\sigma)^2=N(c)\,I$ and $\det(m_c\sigma)=-N(c)$.
\item $\det(C_{ns})=\F_q^\times$: the norm $\F_q[\sqrt\varepsilon]^\times\to\F_q^\times$ is surjective, with kernel of order $q+1$.
\item $C_s\cap\SL_2\simeq C_{q-1}$, $\#(N_s\cap\SL_2)=2(q-1)$, $C_{ns}\cap\SL_2\simeq C_{q+1}$, $\#(N_{ns}\cap\SL_2)=2(q+1)$.
\item The orbits on $\PP^1(\F_q)$ (coordinate $x$ of (C1)) are: for $C_s$, $\{\infty\}$, $\{0\}$, $\F_q^\times$; for $N_s$, $\{0,\infty\}$, $\F_q^\times$; for $\mathrm B$, $\{\infty\}$, $\F_q$.
The group $C_{ns}$ acts simply transitively on $\F_q^2\setminus\{0\}$, and $C_{ns}/\F_q^\times I\simeq C_{q+1}$ acts simply transitively on $\PP^1(\F_q)$; hence $C_{ns}$ and $N_{ns}$
are transitive on $\PP^1(\F_q)$ and stabilise no line of $\F_q^2$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The cosets $wC_s$ and $C_{ns}\sigma$ are disjoint from $C_s$ and $C_{ns}$, so the normalisers have twice the order; the numbers follow (and were checked by
enumeration). (ii) $wC_s$ consists of the matrices $\sm0bc0$, and $m_c\sigma=\sm a{-\varepsilon b}b{-a}$ has trace $0$. The squares: $\sm0bc0^2=\sm{bc}00{bc}$, and
$(m_c\sigma)^2=m_c\,\sigma m_c\sigma^{-1}\,\sigma^2=m_cm_{\bar c}=m_{c\bar c}=N(c)I$. The determinants are immediate. (iii) $N(c)=c\bar c=c^{q+1}$ by
\Cref{prop:groups:cartan}(iii). On the cyclic group $\F_q[\sqrt\varepsilon]^\times$ of order $q^2-1$ the $(q+1)$-th power map has kernel of order $\gcd(q+1,q^2-1)=q+1$ and
image of order $q-1$ (\Cref{thm:groups:cyclic}(ii)), contained in $\F_q^\times$ (as $N(c)=a^2-\varepsilon b^2\in\F_q$), which has order $q-1$. (iv) $C_s\cap\SL_2=\{\mathrm{diag}(a,a^{-1})\}\simeq\F_q^\times$;
in $wC_s$ the determinant-one elements $\sm0bc0$ have $bc=-1$, and there are $q-1$ of them. $C_{ns}\cap\SL_2$ is the kernel of the norm, cyclic (as a subgroup of a
cyclic group) of order $q+1$; in $C_{ns}\sigma$ the determinant-one elements are the $m_c\sigma$ with $N(c)=-1$, a fibre of the surjective norm, with $q+1$ elements.
(v) $\mathrm{diag}(a,d)$ acts by $x\mapsto ax/d$, fixing $0$ and $\infty$ and acting transitively on $\F_q^\times$ (take $a=x'/x$, $d=1$); $w$ acts by $x\mapsto1/x$, swapping $0$ and
$\infty$; $\sm ab0d$ acts by $x\mapsto(ax+b)/d$, fixing $\infty$ and transitive on $\F_q$. Identifying $\F_q^2$ with $\F_q[\sqrt\varepsilon]$ through the basis $(1,\sqrt\varepsilon)$,
$C_{ns}$ acts by multiplication by the nonzero elements of a field, which is simply transitive on the nonzero elements. If $m_c$ stabilises the line $\F_qv$, then $cv=\lambda v$
with $\lambda\in\F_q$, so $(c-\lambda)v=0$ and $c=\lambda$: the stabiliser of every line is $\F_q^\times I$. By \Cref{thm:groups:orbitstab} the orbit of a line has
$(q^2-1)/(q-1)=q+1$ elements, which is all of $\PP^1(\F_q)$, and $C_{ns}/\F_q^\times I$ (cyclic of order $q+1$) acts simply transitively. Since $q+1\ge2$, no line is fixed.
\end{proof}

\begin{theorem}[Dickson]\label{thm:groups:dickson}
Let $\ell$ be an odd prime, $G\le\GL_2(\F_\ell)$ a subgroup and $\bar G$ its image in $\PGL_2(\F_\ell)$.
\begin{enumerate}[label=(\alph*)]
\item If $\ell\mid\#G$, then either $G\supseteq\SL_2(\F_\ell)$ or $G$ is contained in a Borel subgroup.
\item If $\ell\nmid\#G$, then exactly one of the following holds: $\bar G$ is cyclic and $G$ is contained in a Cartan subgroup (split or nonsplit); $\bar G$ is
dihedral of order $2n$ with $n\ge2$, and $G$ is contained in the normaliser of a Cartan subgroup but not in the Cartan subgroup; $\bar G$ is isomorphic to $A_4$,
$S_4$ or $A_5$ (the \emph{exceptional} case).
\end{enumerate}
Consequently a subgroup of $\GL_2(\F_\ell)$ not containing $\SL_2(\F_\ell)$ is contained in a Borel subgroup, in the normaliser of a split or a nonsplit Cartan
subgroup, or in the preimage of an exceptional subgroup of $\PGL_2(\F_\ell)$.
\end{theorem}

We do not prove this; see \cite[Section 2, Propositions 15 and 16]{Serre72} and, for the original classification of the subgroups of $\PSL_2(\F_q)$,
\cite{Dickson1901}. For $\ell=3$ the part needed in the paper is proved below (\Cref{prop:groups:gl23subgroups} and \Cref{thm:groups:threeimages}). Its
statement for $\PGL_2$ and its use for representations is in \Cref{ch:reps}, and the list of the images that occur for elliptic curves over $\Q$ is in
\Cref{ch:galrep}.

\begin{warning}\label{warn:groups:maximal}
\Cref{thm:groups:dickson} is a list of containments, not a list of maximal subgroups. For $\ell=3$ the standard models satisfy $N_s(\F_3)\subset N_{ns}(\F_3)$, with index
$2$ (\Cref{prop:groups:ns3}(vii)). For $\ell=5$, $N_s(\F_5)$ (order $32$, a Sylow $2$-subgroup of $\GL_2(\F_5)$) lies in a subgroup of order $96$ whose image in
$\PGL_2(\F_5)$ is isomorphic to $S_4$ (checked by enumeration; see the proof sketch of \Cref{thm:groups:smallpgl}(iii)). So $N_s(\F_\ell)$ must not be called maximal for
$\ell=3$ or $\ell=5$. The two cases of Proposition prop:ns are distinguished by orbits and by determinant-one subgroups, not by maximality.
\end{warning}

\section{The group \texorpdfstring{$\GL_2(\F_3)$}{GL2(F3)} in full, and \texorpdfstring{$\GL_2(\F_5)$}{GL2(F5)}, \texorpdfstring{$\GL_2(\F_7)$}{GL2(F7)}}\label{sec:groups:gl23}

This section is the group-theoretic core of Proposition prop:ns and of the boundary case studied in Section sec:anomaly. We determine the
conjugacy classes and the subgroups of $\GL_2(\F_3)$, identify $N_s(\F_3)$, $N_{ns}(\F_3)$ and their determinant-one subgroups, and prove that the
three properties used in the paper (surjective determinant, no stable line, an element of determinant $-1$ and trace $0$) leave exactly the
three images $\GL_2(\F_3)$, $N_{ns}(\F_3)$, $N_s(\F_3)$ (\Cref{thm:groups:threeimages}), distinguished by their orbits on the four lines
(\Cref{cor:groups:orbitdichotomy}). The Galois-theoretic side (that the image of complex conjugation has determinant $-1$ and trace $0$, that the
roots of the $3$-division polynomial correspond to lines, and the list of images that occur) is in \Cref{ch:galois}, \Cref{ch:ellcurves} and
\Cref{ch:galrep}.

\emph{Notation for this section.} $J=\sm0{-1}10$ (so $J^2=-I$), $\sigma=\mathrm{diag}(1,-1)$, $w=\sm0110=J\sigma$, $u=\sm1101$. The paper
(Proposition prop:ns and Section sec:anomaly) writes $J$ for $\sm01{-1}0$, which is our $J^{-1}=-J$; both generate the same cyclic group $\langle J\rangle$ of order
$4$, and every statement about the group is unaffected, but a reader comparing formulas line by line should keep the two symbols apart. The four lines of $\F_3^2$ are
$\infty=[1:0]$, $0=[0:1]$, $1=[1:1]$ and $-1=[-1:1]$ (convention (C1)).

\begin{proposition}[Conjugacy classes of $\GL_2(\F_3)$]\label{prop:groups:gl23classes}
$\GL_2(\F_3)$ has order $48$, centre $\{\pm I\}$, commutator subgroup $\SL_2(\F_3)$, and exactly $8$ conjugacy classes:
\begin{center}
\small
\begin{tabular}{cccccccc}
\toprule
representative & size & order & $\det$ & $\tr$ & char.\ polynomial & lines fixed & image in $S_4$\\
\midrule
$I$ & $1$ & $1$ & $1$ & $-1$ & $(X-1)^2$ & $4$ & identity\\
$-I$ & $1$ & $2$ & $1$ & $1$ & $(X+1)^2$ & $4$ & identity\\
$\sigma$ & $12$ & $2$ & $-1$ & $0$ & $X^2-1$ & $2$ & transposition\\
$u$ & $8$ & $3$ & $1$ & $-1$ & $(X-1)^2$ & $1$ & $3$-cycle\\
$J$ & $6$ & $4$ & $1$ & $0$ & $X^2+1$ & $0$ & double transposition\\
$-u$ & $8$ & $6$ & $1$ & $1$ & $(X+1)^2$ & $1$ & $3$-cycle\\
$\sm1{-1}11$ & $6$ & $8$ & $-1$ & $-1$ & $X^2+X-1$ & $0$ & $4$-cycle\\
$\sm0111$ & $6$ & $8$ & $-1$ & $1$ & $X^2-X-1$ & $0$ & $4$-cycle\\
\bottomrule
\end{tabular}
\end{center}
The class of a non-scalar element is determined by its characteristic polynomial. The numbers of elements of orders $1,2,3,4,6,8$ are
$1,13,8,6,8,12$.
\end{proposition}

\begin{proof}
The order is \Cref{prop:groups:orders} and the centre is \Cref{prop:groups:gl2}(i). By \Cref{lem:groups:companion} the non-scalar classes correspond to the
characteristic polynomials $X^2-tX+d$ with $t\in\F_3$, $d\in\{\pm1\}$, all six of which occur (companion matrices), and there are two scalar classes; that makes
$8$. The polynomial $X^2-1$ has distinct roots $\pm1$; $(X\mp1)^2$ are the two with a double root; and $X^2+1$, $X^2+X-1$, $X^2-X-1$ have no root in $\F_3$
(evaluate at $0,1,-1$), so they are irreducible. By \Cref{lem:groups:companion} the centralisers have orders $4$, $6$, $8$ in these three cases, giving class sizes
$12$, $8$, $6$; and $1+1+12+8+6+8+6+6=48$. The representatives listed have the stated characteristic polynomials. Orders: $\sigma^2=I$; $u$ and $-u$ have orders
$3$ and $6$ by \Cref{prop:groups:unipotent}(ii),(iv); $J^2=-I$. An element $g$ with irreducible characteristic polynomial generates a subgroup of
$\F_3[g]^\times$, a cyclic group of order $8$ (it is the unit group of a field with $9$ elements, \Cref{cor:groups:cyclic}); if $g^4=1$, its eigenvalues in that field
satisfy $\lambda^4=1$, so $\lambda\in\{\pm1,\pm\sqrt{-1}\}$, and as $\lambda\notin\F_3$ the characteristic polynomial is $X^2+1$. So the last two classes consist of elements of
order $8$. The lines fixed are the eigenlines over $\F_3$: two for $\sigma$, one for $\pm u$ (\Cref{prop:groups:unipotent}), none for an irreducible characteristic
polynomial, all four for $\pm I$. The image in $S_4$ (\Cref{thm:groups:smallpgl}(i)) has order equal to the order of the element modulo $\{\pm I\}$, and its number of
fixed points is the number of fixed lines: order $2$ with two fixed points is a transposition, order $3$ is a $3$-cycle, order $2$ with no fixed point
($J$, since $J^2=-I$) is a double transposition, and order $4$ (for elements $g$ of order $8$, $g^4=-I$, $g^2$ not scalar) is a $4$-cycle. Commutator subgroup: it lies
in $\SL_2(\F_3)$, as $\det$ kills commutators; and $\sigma u\sigma^{-1}u^{-1}=u^{-2}=u$ is a commutator, so the normal subgroup $[\GL_2(\F_3),\GL_2(\F_3)]$ contains the
whole class of $u$, which generates $\SL_2(\F_3)$ (\Cref{prop:groups:unipotent}(v)). The order statistics are read off the table.
\end{proof}

\begin{proof}

so it suffices to close up all pairs). The rows that the paper needs are proved in \Cref{prop:groups:borel}(iv), \Cref{prop:groups:unipotent},

$8$ elements of order $3$, hence $\SL_2(\F_3)$ by \Cref{prop:groups:unipotent}(v).

\begin{proposition}[$\SL_2(\F_3)$ and $Q_8$]\label{prop:groups:sl23}
\begin{enumerate}[label=(\roman*)]
\item $\SL_2(\F_3)$ has order $24$, and $1,1,8,6,8$ elements of orders $1,2,3,4,6$; $-I$ is its unique element of order $2$.
\item The $8$ elements of $\SL_2(\F_3)$ of order dividing $4$ form the subgroup $Q_8$ of \Cref{conv:groups:models}. It is the unique Sylow $2$-subgroup of
$\SL_2(\F_3)$, it is normal in $\GL_2(\F_3)$, and it is isomorphic to the quaternion group, with $i=J$, $j=\sm111{-1}$, $k=ij=\sm{-1}111$.
\item $\SL_2(\F_3)=Q_8\rtimes\mathrm U\simeq Q_8\rtimes C_3$, where $u$ permutes $i,j,k$ cyclically: $uiu^{-1}=j$, $uju^{-1}=k$, $uku^{-1}=i$.
\item $[\SL_2(\F_3),\SL_2(\F_3)]=Q_8$, and $\SL_2(\F_3)$ has no subgroup of order $12$.
\item $\SL_2(\F_3)/\{\pm I\}\simeq A_4$, $\SL_2(\F_3)\not\simeq S_4$, and $\GL_2(\F_3)/Q_8\simeq S_3$.
\item $Q_8$ acts transitively on the four lines of $\F_3^2$; its image in $S_4$ is the normal subgroup $V_4$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The classes of $\GL_2(\F_3)$ of determinant $1$ are those of $I,-I,u,J,-u$ (\Cref{prop:groups:gl23classes}), of sizes $1,1,8,6,8$; and \Cref{prop:groups:gl2}(iv).
(ii) The elements of order dividing $4$ are $\pm I$ and the six elements of the class of $J$, which are the six matrices of trace $0$ and determinant $1$; these are
$\pm J$, $\pm\sm111{-1}$, $\pm\sm1{-1}{-1}{-1}$, the set $Q_8$ of \Cref{conv:groups:models}. A Sylow $2$-subgroup $P$ of $\SL_2(\F_3)$ has order $8$, and its elements
have order dividing $8$, hence dividing $4$ (there is no element of order $8$ in $\SL_2(\F_3)$); so $P\subseteq Q_8$, and $P=Q_8$ by counting. Hence $Q_8$ is a subgroup, and the
unique Sylow $2$-subgroup. It is a union of conjugacy classes of $\GL_2(\F_3)$ (conjugation preserves order and determinant), hence normal in $\GL_2(\F_3)$. With
$i=J$ and $j=\sm111{-1}$ one computes $k=ij=\sm{-1}111$, $ji=\sm1{-1}{-1}{-1}=-k$; so $Q_8$ is non-abelian with unique involution $-I$, and is the quaternion group by
\Cref{prop:groups:quaternion}(v). (iii) $Q_8$ is normal, $\mathrm U\cap Q_8=1$ (orders $3$ and $8$), and $\#Q_8\mathrm U=24$; apply \Cref{prop:groups:semidirect}(i). Direct
computation with $u^{-1}=\sm1{-1}01$ gives $uJu^{-1}=\sm1{-2}1{-1}=\sm111{-1}=j$ and $uju^{-1}=\sm2{1}1{-2}=\sm{-1}111=k$, and then $uku^{-1}=(uiu^{-1})(uju^{-1})=jk=i$.
(iv) $\SL_2(\F_3)/Q_8\simeq C_3$ is abelian, so the commutator subgroup lies in $Q_8$. It contains $uiu^{-1}i^{-1}=ji^{-1}=-ji=k$, and, being normal, the conjugates
$uku^{-1}=i$ and $uiu^{-1}=j$; so it is $Q_8$. A subgroup of order $12$ would have index $2$, so it would contain every square of $\SL_2(\F_3)$: the $8$ elements of order $3$
(each is the square of its own square, $x=x^4=(x^2)^2$), $-I=J^2$ and therefore the $8$ elements $-u=(-I)u$; that is at least $18$ elements, a contradiction.
(v) The first isomorphism is \Cref{thm:groups:smallpgl}(i). $S_4$ has nine elements of order $2$ (six transpositions, three double transpositions) and
$\SL_2(\F_3)$ has one. $\GL_2(\F_3)/Q_8$ has order $6$; it is generated by the images of $u$ (order $3$) and $\sigma$ (order $2$, as $\sigma\notin Q_8$), with
$\sigma u\sigma^{-1}=u^{-1}$; so it is $D_3\simeq S_3$ by \Cref{prop:groups:dihedral}(i),(iv). (vi) $J$ acts by $x\mapsto-1/x$, that is as $(\infty\ 0)(1\ {-1})$; $j$ acts by
$x\mapsto(x+1)/(x-1)$, that is as $(\infty\ 1)(0\ {-1})$; their product $k$ acts as $(\infty\ {-1})(0\ 1)$. So the image of $Q_8$ is
$\{1,(\infty\,0)(1\,{-1}),(\infty\,1)(0\,{-1}),(\infty\,{-1})(0\,1)\}$, the normal Klein four-group of $S_4$, which is transitive.
\end{proof}

\begin{proposition}[The normaliser of the split Cartan subgroup]\label{prop:groups:ns3}
\begin{enumerate}[label=(\roman*)]
\item The group $N_s(\F_3)=\{\pm I,\pm J,\pm\sigma,\pm w\}$, which equals $\langle J,\sigma\rangle$, has order $8$ and is dihedral, $N_s(\F_3)\simeq D_4$, with $r=J$, $s=\sigma$. Its elements of
determinant $-1$ are exactly $\pm\sigma$, $\pm w$; they have trace $0$ and order $2$.
\item It is the normaliser of $C_s(\F_3)$ and the stabiliser of the pair of lines $\{\infty,0\}$; it has $3$ conjugates, one for each partition of the four lines into
two pairs.
\item Its determinant-one subgroup is $N_s(\F_3)\cap\SL_2(\F_3)=\langle J\rangle\simeq C_4$, cyclic, generated by $J$ and also by the paper's $-J=\sm01{-1}0$.
\item Its orbits on the four lines are $\{\infty,0\}$ (the two lines of the Cartan subgroup) and $\{1,-1\}$ (the other two), each of size $2$; $\langle J\rangle$ has
the same two orbits.
\item $N_s(\F_3)$ stabilises no line of $\F_3^2$ and no line of $\Fbar_3^2$. The group $\langle J\rangle$ stabilises no line of $\F_3^2$, but stabilises the two lines of
$\F_9^2$ spanned by $(i,1)$ and $(-i,1)$, where $i\in\F_9$, $i^2=-1$; these are its eigenlines, with eigenvalues $i$ and $-i$, and $\sigma$ interchanges them.
\item Its image in $\PGL_2(\F_3)\simeq S_4$ is the non-normal Klein four-group generated by the transpositions $(\infty\,0)$ and $(1\,{-1})$.
\item $N_s(\F_3)$ is contained, with index $2$, in the Sylow $2$-subgroup $N_{ns}(\F_3)$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $C_s(\F_3)=\{\pm I,\pm\sigma\}$, and $wC_s(\F_3)=\{\pm w,\pm w\sigma\}=\{\pm w,\pm J\}$ (as $w\sigma=\sm0{-1}10=J$). So $N_s(\F_3)=\{\pm I,\pm J,\pm\sigma,\pm w\}$. Now $J$ has order $4$,
$\sigma$ order $2$, $\sigma J\sigma^{-1}=\sm01{-1}0=J^{-1}$, and $\langle J,\sigma\rangle\subseteq N_s(\F_3)$ has order at least $8$ (it contains $\langle J\rangle$ and $\sigma\notin\langle J\rangle$);
so $N_s(\F_3)=\langle J,\sigma\rangle\simeq D_4$ by \Cref{prop:groups:dihedral}(i). The determinants of $\pm I,\pm J$ are $1$ and those of $\pm\sigma,\pm w$ are $-1$; the
latter have trace $0$ and square to $I$. (ii) is \Cref{prop:groups:cartan}(i) (here $\infty=[e_1]$ and $0=[e_2]$). The stabiliser of the $2$-subset $\{\infty,0\}$ also
stabilises its complement $\{1,-1\}$, and by (iv) $N_s(\F_3)$ stabilises no other $2$-subset. $\GL_2(\F_3)$ is transitive on the six $2$-subsets
(\Cref{prop:groups:projline}(v)), and $gN_s(\F_3)g^{-1}$ is the stabiliser of $g\{\infty,0\}$; so the conjugates correspond to the three partitions into two pairs.
(iii) The determinant-one elements are $\pm I,\pm J$. (iv) $\sigma$ acts by $x\mapsto-x$, as $(1\,{-1})$; $w$ by $x\mapsto1/x$, as $(\infty\,0)$ (note $1/(-1)=-1$); $J$ by
$x\mapsto-1/x$, as $(\infty\,0)(1\,{-1})$. (v) A stable line of $\F_3^2$ would be a fixed point on the four lines, and there is none by (iv). Over $\Fbar_3$:
$[N_s(\F_3),N_s(\F_3)]\ni J\sigma J^{-1}\sigma^{-1}=J\cdot J=-I$, so the commutator subgroup contains an element of order $2$ and is not a $3$-group; apply
\Cref{lem:groups:borelcommutator}(ii). For $\langle J\rangle$: $X^2+1$ has no root in $\F_3$, so $J$ has no eigenline over $\F_3$. Over $\F_9$, $J\binom xy=\binom{-y}x$, and
$\binom{\lambda}1$ is an eigenvector with eigenvalue $\lambda$ iff $\lambda^2=-1$; so the eigenlines are spanned by $(i,1)$ and $(-i,1)$, and they are stable under $\langle J\rangle$.
Finally $\sigma\binom i1=\binom i{-1}=-\binom{-i}1$. (vi) follows from (iv). (vii) $J=m_i\in C_{ns}(\F_3)$ and $\sigma\in N_{ns}(\F_3)$, so $\langle J,\sigma\rangle\subseteq N_{ns}(\F_3)$, which has
order $16$ and is a Sylow $2$-subgroup by \Cref{prop:groups:nns3}(iii).
\end{proof}

\begin{proposition}[The normaliser of the nonsplit Cartan subgroup]\label{prop:groups:nns3}
\begin{enumerate}[label=(\roman*)]
\item The group $C_{ns}(\F_3)$ is cyclic of order $8$, generated by $a=\sm1{-1}11=m_{1+i}$; $a^2=-J$ and $a^4=-I$.
\item $N_{ns}(\F_3)=\langle a,\sigma\rangle$ has order $16$, $\sigma a\sigma^{-1}=a^3$ (conjugation by $\sigma$ is the Frobenius $c\mapsto c^3$ of $\F_9$), and
$N_{ns}(\F_3)\simeq SD_{16}$; it has $1,5,6,4$ elements of orders $1,2,4,8$.
\item It is a Sylow $2$-subgroup of $\GL_2(\F_3)$, with $3$ conjugates.
\item Its determinant-one subgroup is $Q_8$: the four elements $\pm I,\pm J$ of $C_{ns}(\F_3)\cap\SL_2(\F_3)$ and the four elements $m_c\sigma$ with $N(c)=-1$. Its three
subgroups of index $2$ are $C_{ns}(\F_3)\simeq C_8$, $N_s(\F_3)=\langle a^2,\sigma\rangle\simeq D_4$ and $Q_8=\langle a^2,a\sigma\rangle$.
\item Its elements of determinant $-1$ and trace $0$ are the four involutions $m_c\sigma$ with $N(c)=1$, namely $\pm\sigma$, $\pm w$ (the same four as in $N_s(\F_3)$).
$C_{ns}(\F_3)$ contains none: its elements of determinant $-1$ are its four elements of order $8$, of trace $\pm1$.
\item $C_{ns}(\F_3)$, hence $N_{ns}(\F_3)$, acts transitively on the four lines, $C_{ns}(\F_3)/\{\pm I\}\simeq C_4$ simply transitively; the image of $N_{ns}(\F_3)$ in $S_4$
is a Sylow $2$-subgroup, isomorphic to $D_4$. $Q_8$ is also transitive.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $C_{ns}(\F_3)\simeq\F_9^\times$ is cyclic of order $8$ (\Cref{prop:groups:cartan}(ii)). In $\F_9$, $(1+i)^2=2i=-i$ and $(1+i)^4=(-i)^2=-1$, so $1+i$ has order $8$; and
$m_{1+i}=I+J=a$, $a^2=m_{-i}=-J$, $a^4=-I$. (ii) By \Cref{prop:groups:cartan}(iii), $\sigma a\sigma^{-1}=m_{1-i}$ and $(1+i)^3=(1+i)(-i)=1-i$; so $\sigma a\sigma^{-1}=a^3$. The group
$\langle a,\sigma\rangle$ contains $C_{ns}(\F_3)$ and $\sigma$, so it is $N_{ns}(\F_3)$, of order $16$; by \Cref{prop:groups:semidihedral}(iv) it is $SD_{16}$, and (ii) of that
proposition gives the element orders. (iii) $48=16\cdot3$. The number of conjugates is $n_2\in\{1,3\}$, and $N_{ns}(\F_3)$ is not normal: its elements are the matrices
$\sm a{-b}ba$ and $m_c\sigma=\sm ab{b}{-a}$, and $u\sigma u^{-1}=\sm11{0}{-1}$ is of neither form. So $n_2=3$. (iv) $C_{ns}\cap\SL_2$ is the kernel of the norm, of order $4$, which
is $\langle a^2\rangle=\langle J\rangle$; in the other coset $\det(m_c\sigma)=-N(c)$, and $N(c)=-1$ for four values of $c$ (\Cref{prop:groups:cartanfacts}(iv)). So the
determinant-one subgroup has order $8$; it is a $2$-subgroup of $\SL_2(\F_3)$ of order $8$, hence the Sylow $2$-subgroup $Q_8$ (\Cref{prop:groups:sl23}(ii)). The index-$2$
subgroups are those of \Cref{prop:groups:semidihedral}(iii): $\langle a\rangle=C_{ns}(\F_3)$; $\langle a^2,\sigma\rangle=\langle -J,\sigma\rangle=N_s(\F_3)$; and $\langle a^2,a\sigma\rangle$, which lies in $\SL_2(\F_3)$
($\det a=\det\sigma=-1$) and has order $8$, so it is $Q_8$. (v) The elements $m_c\sigma$ have trace $0$, and determinant $-1$ iff $N(c)=1$, that is $c\in\{\pm1,\pm i\}$; these give
$\pm\sigma$ and $\pm J\sigma=\pm w$. In $C_{ns}(\F_3)$, $\det a=-1$ and $\det a^2=1$, so the elements of determinant $-1$ are the odd powers of $a$, the elements of order $8$;
$a$ and $a^3=m_{1-i}$ have trace $2=-1$, and $a^5=-a$, $a^7=-a^3$ have trace $1$. (vi) is \Cref{prop:groups:cartanfacts}(v) for $q=3$, and the image has order $16/2=8$, a Sylow
$2$-subgroup of $S_4$, which is dihedral (\Cref{ex:groups:sylowsmall}(b)). For $Q_8$ see \Cref{prop:groups:sl23}(vi).
\end{proof}

\begin{theorem}[The three images]\label{thm:groups:threeimages}
Let $H\le\GL_2(\F_3)$ satisfy:
\begin{enumerate}[label=(\alph*)]
\item $\det(H)=\F_3^\times$;
\item $H$ stabilises no line of $\F_3^2$;
\item $H$ contains an element of determinant $-1$ and trace $0$.
\end{enumerate}
Then $H$ is conjugate to exactly one of $\GL_2(\F_3)$, $N_{ns}(\F_3)$ and $N_s(\F_3)$, of orders $48$, $16$, $8$. Their determinant-one subgroups $H\cap\SL_2(\F_3)$,
of index $2$, are $\SL_2(\F_3)$, $Q_8$ and $C_4=\langle J\rangle$ respectively. The first two stabilise no line of $\Fbar_3^2$; the third is abelian and stabilises two lines of
$\F_9^2$. The subgroup $C_{ns}(\F_3)$ satisfies (a) and (b) but not (c).
\end{theorem}

\begin{proof}
The three conditions are invariant under conjugation, and the three groups satisfy them: (a) and (c) because each contains $\sigma$; (b) because $\GL_2(\F_3)$ and
$N_{ns}(\F_3)$ are transitive on the four lines and $N_s(\F_3)$ has two orbits of size $2$ (\Cref{prop:groups:nns3}(vi), \Cref{prop:groups:ns3}(iv)). They have different
orders, so $H$ is conjugate to at most one of them.

Suppose first that $3\mid\#H$. By \Cref{prop:groups:gl23subgroups} (last statement) and (b), $H\supseteq\SL_2(\F_3)$; by (a), $H$ contains an element of determinant $-1$, so
$H$ has index at most $2$ in $\GL_2(\F_3)$ and properly contains $\SL_2(\F_3)$, whence $H=\GL_2(\F_3)$.

Suppose now that $3\nmid\#H$. Then $\#H$ divides $16$, $H$ is a $2$-group, and by \Cref{thm:groups:sylow}(ii) it is contained in a Sylow $2$-subgroup, which is conjugate to
$N_{ns}(\F_3)$. Replacing $H$ by a conjugate we may assume $H\le N_{ns}(\F_3)$. By (a), $H\cap\SL_2(\F_3)$ has index $2$ in $H$; in particular $\#H\ge2$.
\begin{itemize}
\item If $\#H=16$, then $H=N_{ns}(\F_3)$.
\item If $\#H=8$, $H$ is one of the three subgroups of index $2$ of \Cref{prop:groups:nns3}(iv). $Q_8$ fails (a), $C_{ns}(\F_3)$ fails (c) by
\Cref{prop:groups:nns3}(v), and $N_s(\F_3)$ satisfies all three. So $H=N_s(\F_3)$.
\item If $\#H\in\{2,4\}$, then by (c) $H$ contains an element $g$ of determinant $-1$ and trace $0$, an involution with two eigenlines (characteristic polynomial
$X^2-1$). Then $H\cap\SL_2(\F_3)$ has order $\#H/2\le2$, so it is $1$ or $\{\pm I\}$ ($-I$ being the only involution of $\SL_2(\F_3)$), and $H=(H\cap\SL_2(\F_3))\cup g(H\cap\SL_2(\F_3))$
is $\{I,g\}$ or $\{\pm I,\pm g\}$. Both stabilise the eigenlines of $g$, contradicting (b).
\end{itemize}
So $H$ is conjugate to $\GL_2(\F_3)$, $N_{ns}(\F_3)$ or $N_s(\F_3)$. The determinant-one subgroups are computed in \Cref{prop:groups:nns3}(iv) and \Cref{prop:groups:ns3}(iii).
The commutator subgroups of $\SL_2(\F_3)$ and of $Q_8$ are $Q_8$ and $\{\pm I\}$ (\Cref{prop:groups:sl23}(iv), \Cref{prop:groups:quaternion}(ii)), which are not $3$-groups, so by
\Cref{lem:groups:borelcommutator}(ii) these groups stabilise no line of $\Fbar_3^2$. The group $\langle J\rangle$ is cyclic and stabilises the two eigenlines of $J$ in $\F_9^2$
(\Cref{prop:groups:ns3}(v)). Finally $C_{ns}(\F_3)$ has surjective determinant and is transitive on the lines (\Cref{prop:groups:cartanfacts}(iii),(v)), but contains no
element of determinant $-1$ and trace $0$ (\Cref{prop:groups:nns3}(v)).
\end{proof}

\begin{corollary}[Orbit dichotomy]\label{cor:groups:orbitdichotomy}
For $H$ as in \Cref{thm:groups:threeimages} the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $H$ is conjugate to $N_s(\F_3)$;
\item $H$ has two orbits, each of size $2$, on the four lines of $\F_3^2$;
\item $H$ is not transitive on the four lines;
\item $H\cap\SL_2(\F_3)$ is abelian;
\item $H\cap\SL_2(\F_3)$ stabilises a line of $\Fbar_3^2$.
\end{enumerate}
Otherwise $H$ is transitive on the four lines.
\end{corollary}

\begin{proof}
All five properties are invariant under conjugation, so by \Cref{thm:groups:threeimages} we may take $H$ to be one of the three standard models. For $N_s(\F_3)$, (ii) holds
by \Cref{prop:groups:ns3}(iv), (iv) because $\langle J\rangle$ is cyclic, and (v) by \Cref{prop:groups:ns3}(v); (ii) implies (iii). For $\GL_2(\F_3)$ and $N_{ns}(\F_3)$, (ii) and (iii)
fail because they are transitive, (iv) fails because $\SL_2(\F_3)$ and $Q_8$ are non-abelian, and (v) fails by \Cref{thm:groups:threeimages}.
\end{proof}

\begin{proposition}[Determinant $-1$ and trace $0$]\label{prop:groups:cc}
Let $q$ be odd and $g\in\GL_2(\F_q)$. Then $\det g=-1$ and $\tr g=0$ if and only if the characteristic polynomial of $g$ is $X^2-1$, if and only if $g$ is conjugate to
$\sigma=\mathrm{diag}(1,-1)$. Such an element has order $2$ and fixes exactly two lines (its eigenlines); for $q=3$ it swaps the other two. These elements form one
conjugacy class of $q(q+1)$ elements. For $q=3$ their numbers in the standard subgroups are: $\GL_2(\F_3)$: $12$; $\mathrm B$: $6$; $N_s$: $4$; $N_{ns}$: $4$; $C_s$: $2$; $C_{ns}$: $0$;
$\SL_2(\F_3)$: $0$.
\end{proposition}

\begin{proof}
The characteristic polynomial is $X^2-\tr(g)X+\det g$. A matrix with characteristic polynomial $X^2-1$ is not scalar (a scalar $\lambda I$ has $(X-\lambda)^2$, and
$1\ne-1$), so it is conjugate to $\sigma$ by \Cref{lem:groups:companion}. Its eigenvalues $1$, $-1$ are distinct, which gives the order and the two eigenlines. For
$q=3$, $\sigma$ acts as the transposition $(1\,{-1})$ (\Cref{prop:groups:gl23classes}). The centraliser of $\sigma$ is $C_s$, of order $(q-1)^2$, so the class has
$q(q-1)^2(q+1)/(q-1)^2=q(q+1)$ elements. In $\mathrm B(\F_3)$: $\sm ab0d$ with $a+d=0$ and $ad=-1$ means $(a,d)=\pm(1,-1)$ with $b$ arbitrary, $6$ elements. The counts in $N_s$,
$N_{ns}$ and $C_{ns}$ are \Cref{prop:groups:ns3}(i) and \Cref{prop:groups:nns3}(v); in $C_s$ they are $\pm\sigma$; and $\SL_2$ has no element of determinant $-1$.
\end{proof}

\begin{example}[Sutherland's labels at $\ell=3$]\label{ex:groups:sutherland3}
The paper names mod $3$ images by the labels of \cite{Sutherland16}. Because a transposition (row vectors versus column vectors) interchanges some of the
matrix descriptions, we define each label by an intrinsic property of the action on $E[3]$, which is convention-free; the groups are then the standard models
up to conjugacy, and the orders are those of \Cref{prop:groups:gl23subgroups}.
\begin{itemize}
\item \textbf{3Cs.1.1}: $\{I,\sigma\}$, of order $2$: $E[3]\simeq\Z/3\oplus\mu_3$, that is, a basis in which $G_\Q$ acts trivially on the first vector and by the
determinant $\bar\chi_3$ on the second. (It is a subgroup of the split Cartan subgroup, not the split Cartan subgroup itself.)
\item \textbf{3Cs}: $C_s(\F_3)$, of order $4$: the full group of matrices stabilising each of two distinct lines.
\item \textbf{3B.1.1}: $\{\sm1*0*\}$, of order $6$: a fixed nonzero vector, that is, a rational point of order $3$.
\item \textbf{3B.1.2}: $\{\sm**01\}$, of order $6$: a stable line, on which the group acts by the determinant, with trivial action on the quotient.
\item \textbf{3B}: $\mathrm B(\F_3)$, of order $12$: the full stabiliser of a line; a stable line means a rational $3$-isogeny.
\item \textbf{3Ns}: $N_s(\F_3)$, of order $8$. \textbf{3Nn}: $N_{ns}(\F_3)$, of order $16$.

\item The surjective image $\GL_2(\F_3)$ carries no proper label.

These are the nine conjugacy classes of subgroups with surjective determinant in \Cref{prop:groups:gl23subgroups}. The table there shows that 3B.1.1 and 3B.1.2 are not

\end{example}

exceptional case $\bar G\simeq S_4$ of \Cref{thm:groups:dickson} occurs inside $\PGL_2(\F_5)\simeq S_5$ and contains the image of $N_s(\F_5)$ (\Cref{warn:groups:maximal}). The
coincidence that makes $\ell=3$ special in the paper is not group-theoretic: for $\ell=3$ the reduction $\bar\chi_3$ of the cyclotomic character equals the quadratic
character $\chi_{-3}$, while for $\ell\ge5$ the character $\bar\chi_\ell$ has order $\ell-1>2$ and differs from $\chi_{\ell^*}$ (\Cref{thm:groups:cyclic}(iv)). This is why
Lemma lem:h0 gives $h_\ell(E)=0$ for $\ell\ge5$ (\Cref{ch:reps}, \Cref{ch:galrep}).
\end{example}

\section{Inverse limits and profinite groups}\label{sec:groups:profinite}

The Galois groups of the paper, $G_\Q=\Gal(\Qbar/\Q)$, the local groups $G_{\Q_p}$ and the inertia groups $I_p$, are infinite, and the representations $\rho_{E,\ell}$
take values in $\GL_2(\Z_\ell)$. Both are profinite groups: inverse limits of finite groups, with a compact topology in which ``continuous'' means ``factors through a
finite quotient at each finite level''. This section gives the definitions and the facts used in the paper: a continuous representation modulo $\ell^k$ factors through a
finite quotient (Lemma lem:h0, Lemma lem:tamagawa-exp), a pro-$p$ group has no nontrivial continuous image of order prime to $p$ (the wild inertia in Lemma
lem:tamfactor), and the kernel of $\GL_2(\Z_3)\to\GL_2(\F_3)$ is torsion-free (Lemma lem:IV). The infinite Galois theory itself is \Cref{ch:galois}.

\begin{definition}[Inverse limits; profinite groups]\label{def:groups:inverselimit}
An \emph{inverse system} of groups (indexed by $\N$) is a sequence of groups $G_n$, $n\ge1$, with homomorphisms $\pi_n\colon G_{n+1}\to G_n$. Its \emph{inverse limit} is
\[
\varprojlim G_n=\Bigl\{(g_n)_{n\ge1}\in\prod_{n\ge1}G_n:\ \pi_n(g_{n+1})=g_n\text{ for all }n\Bigr\},
\]
a subgroup of the product, with projections $p_n\colon\varprojlim G_n\to G_n$. When the $G_n$ are finite, each is given the discrete topology, $\prod G_n$ the product topology
and $\varprojlim G_n$ the subspace topology. A \emph{profinite group} is a topological group isomorphic, as a topological group, to such an inverse limit of finite groups. A
\emph{pro-$p$ group} is an inverse limit of finite $p$-groups.
\end{definition}

\begin{remark}\label{rem:groups:directed}
For Galois groups one needs inverse systems indexed by a \emph{directed} set $(I,\le)$ (a partial order in which any two elements have an upper bound), with maps
$\pi_{ij}\colon G_j\to G_i$ for $i\le j$ satisfying $\pi_{ii}=\id$ and $\pi_{ij}\pi_{jk}=\pi_{ik}$; the limit is the set of $(g_i)\in\prod_iG_i$ with $\pi_{ij}(g_j)=g_i$. Everything
below holds in this generality with the same proofs, replacing ``$n$ large'' by ``$i$ large in $I$''. For example $\Gal(\Qbar/\Q)=\varprojlim\Gal(K/\Q)$ over the finite Galois
extensions $K/\Q$ inside $\Qbar$, ordered by inclusion (\Cref{ch:galois}).
\end{remark}

\begin{proposition}\label{prop:groups:inverselimit}
Let $(G_n,\pi_n)$ be an inverse system of finite groups and $G=\varprojlim G_n$.
\begin{enumerate}[label=(\roman*)]
\item (Universal property) For every group $H$ with homomorphisms $f_n\colon H\to G_n$ such that $\pi_n\circ f_{n+1}=f_n$, there is a unique homomorphism $f\colon H\to G$ with
$p_n\circ f=f_n$ for all $n$.
\item $G$ is a closed subgroup of $\prod G_n$, and a Hausdorff topological group.
\item $G$ is compact.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Put $f(h)=(f_n(h))_n$, which lies in $G$ by the compatibility; it is a homomorphism, and it is the only map with $p_n\circ f=f_n$. (ii) For each $n$ the set
$\{(g_m):\pi_n(g_{n+1})=g_n\}$ is the preimage of the closed subset $\{(x,y):\pi_n(x)=y\}$ of the discrete space $G_{n+1}\times G_n$ under the continuous projection; so it is
closed, and $G$ is the intersection of these closed sets. Multiplication and inversion are continuous coordinatewise, hence continuous on the product and on $G$. A
product of Hausdorff spaces is Hausdorff. (iii) By Tychonoff's theorem a product of compact spaces is compact (we do not prove this; see
\cite[Theorem 37.3]{Munkres00}); finite discrete spaces are compact, and a closed subset of a compact space is compact.
\end{proof}

\begin{example}[The $p$-adic integers]\label{ex:groups:zp}
Let $p$ be a prime. The rings $\Z/p^n$ with the reduction maps form an inverse system, and $\Zp=\varprojlim\Z/p^n$ is a ring (the operations being coordinatewise) and a
compact topological group; its analysis, as the completion of $\Z$ for the $p$-adic absolute value, is \Cref{ch:local}.
\begin{enumerate}[label=(\alph*)]
\item $x=(x_n)\in\Zp$ is a unit if and only if $x_1\ne0$ in $\Z/p$; and $\Zp^\times=\varprojlim(\Z/p^n)^\times$. Indeed, if $x_1\ne0$, every $x_n$ is prime to $p$, hence a unit of
$\Z/p^n$, and the inverses $x_n^{-1}$ are compatible (inverses are unique), giving $x^{-1}$; conversely $x_1$ is a unit if $x$ is.
\item The kernel of $\Zp\to\Z/p^n$ is $p^n\Zp$. If $x_n=0$, then for every $k$ the element $x_{n+k}\in\Z/p^{n+k}$ is divisible by $p^n$, $x_{n+k}=p^ny_k$ with $y_k$ well defined
in $\Z/p^k$, and the $y_k$ are compatible, so $x=p^ny$. The converse is clear.
\item $\Zp$ is torsion-free. If $mx=0$ with $m\ge1$ and $v=v_p(m)$, then $mx_{n+v}=0$ in $\Z/p^{n+v}$, so $p^n\mid x_{n+v}$ (as $m/p^v$ is invertible), and $x_n=x_{n+v}\bmod p^n=0$
for every $n$.
\item The open subgroups of $\Zp$ are the $p^n\Zp$, $n\ge0$. By \Cref{prop:groups:profinite}(i),(ii) (proved below) an open subgroup $V$ contains some $p^n\Zp$ and corresponds to a subgroup of
$\Zp/p^n\Zp\simeq\Z/p^n$, which is $p^k\Z/p^n$ for some $k\le n$ (\Cref{thm:groups:cyclic}(i)); so $V=p^k\Zp$.
\item Let $\widehat\Z=\varprojlim_n\Z/n!$; every $\Z/m$ is a quotient of $\widehat\Z$ (as $m\mid n!$ for $n\ge m$). By \Cref{prop:groups:crtgroups} (in its ring form, \Cref{ch:rings})
$\Z/n!\simeq\prod_p\Z/p^{v_p(n!)}$, compatibly in $n$, and taking limits gives $\widehat\Z\simeq\prod_p\Zp$.
\item (Foretaste.) For a prime $\ell$ and a field $K$ of characteristic $\ne\ell$ containing all $\ell$-power roots of unity, $\Z_\ell(1)=\varprojlim\mu_{\ell^n}(K)$, with the
maps $\zeta\mapsto\zeta^\ell$. Choosing a compatible system of generators $\zeta_{\ell^n}$ ($\zeta_{\ell^{n+1}}^\ell=\zeta_{\ell^n}$) gives an isomorphism $\Z_\ell\simeq\Z_\ell(1)$ of groups, but the
Galois group acts on $\Z_\ell(1)$ through the cyclotomic character $\chi_\ell$ (\Cref{ch:galois}, \Cref{ch:local}). The paper's coefficient modules $\Z_\ell$, $T_\ell E$,
$\Z_\ell(1)$ and $\Qlmod=\varinjlim\ell^{-n}\Z_\ell/\Z_\ell$ are all built from these.
\end{enumerate}
\end{example}

\begin{proposition}[Profinite groups]\label{prop:groups:profinite}
Let $G=\varprojlim G_n$ be profinite, and $U_n=\ker(p_n\colon G\to G_n)$.
\begin{enumerate}[label=(\roman*)]
\item The $U_n$ are open normal subgroups of finite index, $U_{n+1}\subseteq U_n$, $\bigcap_nU_n=1$, and they form a basis of open neighbourhoods of $1$.
\item A subgroup of $G$ is open if and only if it is closed of finite index.
\item A homomorphism $f$ from $G$ to a finite group $F$ that is continuous ($F$ discrete) has open kernel, and factors through $G/U_n$ for some $n$; if the maps $p_n$ are
surjective, $f$ factors through $G_n$.
\item Closed subgroups of $G$, and quotients of $G$ by closed normal subgroups, are profinite. The image of a continuous homomorphism from $G$ to a Hausdorff
topological group (such as $\GL_2(\Z_\ell)$) is closed.
\item A topological group is profinite (for an inverse system indexed by a directed set, \Cref{rem:groups:directed}) if and only if it is compact, Hausdorff and totally
disconnected.
\item Closed subgroups and quotients by closed normal subgroups of a pro-$p$ group are pro-$p$; every continuous homomorphism from a pro-$p$ group to a finite group of
order prime to $p$ is trivial.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $U_n=p_n^{-1}(1)$ is open ($p_n$ is continuous and $G_n$ discrete), normal, and of index at most $\#G_n$. If $g\in U_{n+1}$, then $g_n=\pi_n(g_{n+1})=1$; so $U_{n+1}\subseteq U_n$, and
likewise $g\in U_n$ implies $g_m=1$ for all $m\le n$. An element of all $U_n$ is $1$. A basic open neighbourhood of $1$ in the product restricts finitely many coordinates,
say $g_m\in S_m$ for $m\le n$, with $1\in S_m$; it contains $U_n$ by the previous remark. (ii) If $V$ is an open subgroup, its complement is the union of the open cosets
$gV\ne V$, so $V$ is closed; the cosets of $V$ form a disjoint open cover of the compact space $G$, so there are finitely many. Conversely, if $V$ is closed of finite
index, its complement is a finite union of closed cosets, so $V$ is open. (iii) $\ker f=f^{-1}(1)$ is open, so it contains some $U_n$ by (i), and $f$ factors through
$G/U_n$, which is isomorphic to $p_n(G)\le G_n$ by the first isomorphism theorem, and to $G_n$ if $p_n$ is surjective. (iv) Let $H\le G$ be closed and $H_n=p_n(H)$; the
$H_n$ form an inverse system, and $h\mapsto(p_n(h))$ is a continuous injective homomorphism $H\to\varprojlim H_n$ (injective as $\bigcap U_n=1$). It is surjective: given
compatible $(h_n)$ with $h_n=p_n(x_n)$, $x_n\in H$, the sets $C_n=H\cap x_nU_n$ are closed, nonempty and decreasing (compatibility gives $x_{n+1}U_n=x_nU_n$), so by
compactness $\bigcap C_n\ne\emptyset$, and any $h$ in it maps to $(h_n)$. A continuous bijection from a compact space to a Hausdorff space is a homeomorphism, so
$H\simeq\varprojlim H_n$. For a closed normal $N$, the same argument applied to the finite groups $G/NU_n$ shows $G/N\simeq\varprojlim G/NU_n$; injectivity now needs $\bigcap_nNU_n=N$,
which holds because $N$ is closed: if $g\notin N$, the closed sets $N\cap gU_n$ are decreasing with empty intersection (as $\bigcap_ngU_n=\{g\}$), so one of them is empty
by compactness, that is, $g\notin NU_n$ for some $n$. The
image of a compact space under a continuous map is compact, and compact subsets of Hausdorff spaces are closed. (v) We do not prove this; see \cite[Chapter I,
Section 1]{NSW} and \cite[Chapter IV, Section 2]{Neukirch}. (vi) In the proof of (iv) the groups $H_n\le G_n$ and $G/NU_n$ (a quotient of $G/U_n\simeq p_n(G)$) are $p$-groups
when the $G_n$ are. If $f\colon G\to F$ is continuous with $p\nmid\#F$, then by (iii) $f(G)$ is a quotient of $G/U_n$, a $p$-group, and has order dividing $\#F$; so it is trivial.
\end{proof}

\begin{proposition}[$\GL_2(\Z_\ell)$ and its congruence subgroup]\label{prop:groups:gl2zl}
Let $\ell$ be a prime and $\GL_2(\Z_\ell)$ the group of $2\times2$ matrices over $\Z_\ell$ with determinant in $\Z_\ell^\times$.
\begin{enumerate}[label=(\roman*)]
\item $\GL_2(\Z_\ell)=\varprojlim\GL_2(\Z/\ell^n)$. Each reduction map $\GL_2(\Z/\ell^{n+1})\to\GL_2(\Z/\ell^n)$ ($n\ge1$) is surjective with kernel $\{I+\ell^nX\}\simeq(M_2(\F_\ell),+)$
of order $\ell^4$; so $\#\GL_2(\Z/\ell^n)=\ell^{4(n-1)}(\ell^2-1)(\ell^2-\ell)$. For $\ell=3$, $n=2$: $\#\GL_2(\Z/9)=81\cdot48=3888$.
\item The kernel $\Gamma(\ell)=I+\ell M_2(\Z_\ell)$ of reduction modulo $\ell$ is a pro-$\ell$ group.
\item If $\ell$ is odd, $\Gamma(\ell)$ is torsion-free. Hence every finite subgroup of $\GL_2(\Z_\ell)$ maps isomorphically onto its image in $\GL_2(\F_\ell)$. For $\ell=2$ this
fails: $-I\in\Gamma(2)$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $M_2(\Z_\ell)=\varprojlim M_2(\Z/\ell^n)$ entrywise, and a matrix is invertible over $\Z_\ell$ iff its determinant is a unit, iff the determinant is nonzero modulo $\ell$
(\Cref{ex:groups:zp}(a)), iff all its reductions are invertible. Given $\bar g\in\GL_2(\Z/\ell^n)$, any lift $g$ to $M_2(\Z/\ell^{n+1})$ has $\det g\equiv\det\bar g\not\equiv0\pmod\ell$,
so it is invertible: the reduction is surjective. Its kernel consists of the $I+\ell^nX$, with $X$ well defined modulo $\ell$, and $(I+\ell^nX)(I+\ell^nY)=I+\ell^n(X+Y)+\ell^{2n}XY
\equiv I+\ell^n(X+Y)\pmod{\ell^{n+1}}$, as $2n\ge n+1$; these matrices are invertible (determinant $\equiv1\bmod\ell$). So the kernel is isomorphic to $(M_2(\F_\ell),+)$, and the
order follows by induction from \Cref{prop:groups:orders}. (ii) $\Gamma(\ell)$ is the kernel of the continuous map $p_1$, hence closed, and by the proof of
\Cref{prop:groups:profinite}(iv) it is the inverse limit of its images in $\GL_2(\Z/\ell^n)$, which lie in the kernels of $\GL_2(\Z/\ell^n)\to\GL_2(\F_\ell)$, groups of order
$\ell^{4(n-1)}$. (iii) Let $g\in\Gamma(\ell)$, $g\ne I$, and write $g=I+\ell^kA$ with $k\ge1$ and $A\in M_2(\Z_\ell)$, $A\not\equiv0\pmod\ell$. Then
\[
g^\ell=I+\ell^{k+1}A+\sum_{i=2}^{\ell-1}\binom\ell i\ell^{ki}A^i+\ell^{k\ell}A^\ell .
\]
For $2\le i\le\ell-1$ the term is divisible by $\ell^{1+2k}$, and $k\ell\ge k+2$ because $k(\ell-1)\ge2$ for $\ell\ge3$; so $g^\ell=I+\ell^{k+1}A_1$ with $A_1\equiv A\pmod\ell$.
By induction $g^{\ell^s}=I+\ell^{k+s}A_s$ with $A_s\equiv A\not\equiv0\pmod\ell$. Now let $m\ge1$, $m=\ell^sm'$ with $\ell\nmid m'$, and $h=g^{\ell^s}$. Then
$h^{m'}=I+m'\ell^{k+s}A_s+(\text{terms divisible by }\ell^{2(k+s)})\equiv I+m'\ell^{k+s}A_s\pmod{\ell^{k+s+1}}$, and $m'A_s\not\equiv0\pmod\ell$; so $g^m=h^{m'}\ne I$. Thus $g$
has infinite order. If $H\le\GL_2(\Z_\ell)$ is finite, $H\cap\Gamma(\ell)$ is a finite subgroup of a torsion-free group, hence trivial, and reduction is injective on $H$.
Finally $-I=I+2(-I)$ has order $2$.
\end{proof}

\begin{paperlink}
The proof of Lemma lem:IV uses \Cref{prop:groups:gl2zl}(iii) with $\ell=3$: ``reduction modulo $3$ is injective on it, the kernel of $\GL_2(\Z_3)\to\GL_2(\F_3)$ being
torsion-free'', applied to the finite group $\rho_{E,3}(I_q)$; together with \Cref{prop:groups:unipotent}(iii),(v) this shows that $\rho_{E,3}(I_q)$ is cyclic of order $3$.
Note the compatibility: an element of order $3$ of $\GL_2(\Z_3)$ has eigenvalues the primitive cube roots of unity, characteristic polynomial $X^2+X+1$, and reduces to
an element of characteristic polynomial $X^2+X+1=(X-1)^2$ in $\F_3[X]$, a nontrivial unipotent; the structure of $T_3E$ as a module over $\Z_3[\zeta_3]$ is
\Cref{ch:reps} and \Cref{ch:galrep}. Lemma lem:h0 and Lemma lem:tamagawa-exp consider $\rho\bmod\ell^k$, the composite of $\rho$ with $\GL_2(\Z_\ell)\to\GL_2(\Z/\ell^k)$; it is
a continuous homomorphism to a finite group, so it factors through a finite quotient of $G_\Q$ (\Cref{prop:groups:profinite}(iii)).
\end{paperlink}

\begin{example}[Profinite Galois groups; forward pointers]\label{ex:groups:galoisprofinite}
The following statements are proved in \Cref{ch:galois} and \Cref{ch:local}; we record them to show where the definitions are used.
\begin{enumerate}[label=(\alph*)]
\item $\Gal(\Qbar/\Q)=\varprojlim\Gal(K/\Q)$ over the finite Galois extensions $K/\Q$, a profinite group (\Cref{rem:groups:directed}).
\item $\Gal(\Q(\mu_{\ell^\infty})/\Q)\simeq\Z_\ell^\times$ through the cyclotomic character $\chi_\ell$.
\item $\Gal(\Fbar_q/\F_q)\simeq\widehat\Z$, topologically generated by the Frobenius $x\mapsto x^q$. A profinite group topologically generated by one element (procyclic) is a
quotient of $\widehat\Z$.
\item For a prime $p$, the inertia group $I_p\le G_{\Q_p}$ has a normal pro-$p$ subgroup, the wild inertia group, with quotient the tame inertia group
$\prod_{\ell\ne p}\Z_\ell(1)$, which is procyclic. So a finite image of inertia of order prime to $p$ is a quotient of the tame inertia group, hence cyclic (by
\Cref{prop:groups:profinite}(vi) the wild inertia group maps trivially). This is the step ``For $q\ge5$ the image is tame, hence cyclic'' in the proof of Proposition prop:shapes, and the step ``The order $3$ is
prime to $q$, so the image is a quotient of the tame inertia group'' in the proof of Lemma lem:IV.
\end{enumerate}
\end{example}

\section{Exercises}\label{sec:groups:exercises}

The exercises are graded from routine to harder. Exercises \ref{ex:groups:ex5} to \ref{ex:groups:ex9} check the group-theoretic claims of Proposition prop:ns
directly.

\begin{exercise}\label{ex:groups:ex1}
Show that $\#\GL_2(\Z/9)=3888$ and $\#\SL_2(\Z/9)=648$. (Use \Cref{prop:groups:gl2zl}(i) and the surjectivity of $\det$ onto $(\Z/9)^\times$, a group of order $6$.)
\end{exercise}

\begin{exercise}\label{ex:groups:ex2}
Compute the conjugacy classes of $D_4$ and of $Q_8$: both have five classes, of sizes $1,1,2,2,2$. Deduce from the formula $\#G=\sum_\chi\chi(1)^2$ for the
degrees of the irreducible complex characters (Riley-Hobson-Bence, the chapter on representation theory; \Cref{ch:reps}) that both groups have four characters of degree $1$ and one of
degree $2$, although they are not isomorphic.
\end{exercise}

\begin{exercise}\label{ex:groups:ex3}
Prove that $C_8\times C_2$, $D_8$, $SD_{16}$, $Q_{16}$ and $M_{16}$ (\Cref{rem:groups:order16}) are pairwise non-isomorphic, by counting elements of each order.
\end{exercise}

\begin{exercise}\label{ex:groups:ex4}
Find all subgroups of $SD_{16}$ and identify each up to isomorphism. (There are $15$: the trivial group, five of order $2$, five of order $4$ (three cyclic, two
Klein), three of order $8$, and $SD_{16}$.) Which are normal?
\end{exercise}

\begin{exercise}\label{ex:groups:ex5}
Verify the class table of \Cref{prop:groups:gl23classes} by listing the $48$ elements of $\GL_2(\F_3)$. Then use the orbit-counting lemma
(\Cref{lem:groups:burnside}) and the column ``lines fixed'' to show that $N_s(\F_3)$ has two orbits on $\PP^1(\F_3)$ and that $N_{ns}(\F_3)$ has one.
\end{exercise}

\begin{exercise}\label{ex:groups:ex6}
Check the paper's claims that the determinant-one subgroup of $N_s(\F_3)$ is cyclic of order $4$, generated by $\sm01{-1}0$, and that the determinant-one subgroup of
$N_{ns}(\F_3)$ is the quaternion group.
\end{exercise}

\begin{exercise}\label{ex:groups:ex7}
Check the paper's claim that $N_s(\F_3)$ has two orbits of size two on the four lines of $\F_3^2$ (the two lines of the Cartan subgroup and the two others), while
$N_{ns}(\F_3)$ and $\GL_2(\F_3)$ are transitive. Find an element of $N_{ns}(\F_3)$ that maps $\infty$ to $1$.
\end{exercise}

\begin{exercise}\label{ex:groups:ex8}
Show that $C_{ns}(\F_3)$ contains no element of determinant $-1$ and trace $0$, and conclude that the image of $\bar\rho_{E,3}$ for an elliptic curve over $\Q$ is never
conjugate to $C_{ns}(\F_3)$ (the label 3Cn), using that complex conjugation maps to an element of determinant $-1$ and order $2$.
\end{exercise}

\begin{exercise}\label{ex:groups:ex9}
Show that every subgroup of $\GL_2(\F_3)$ of order $6$ stabilises a line. Classify these subgroups up to conjugacy: two classes isomorphic to $S_3$ (3B.1.1 and 3B.1.2)
and one cyclic class, $\{\pm I\}\times\mathrm U$. Show that 3B.1.1 and 3B.1.2 are interchanged by $g\mapsto(g^{-1})^{\mathsf T}$, which is an automorphism of $\GL_2(\F_3)$ but not an
inner one.
\end{exercise}

\begin{exercise}\label{ex:groups:ex10}
Prove \Cref{thm:groups:smallpgl}(ii) by direct computation: for each of the eight class representatives of \Cref{prop:groups:gl23classes}, compare the determinant
with the sign of the induced permutation of the four lines.
\end{exercise}

\begin{exercise}\label{ex:groups:ex11}
Let $q$ be odd. Show that $N_{ns}(\F_q)\cap\SL_2(\F_q)$ has a unique element of order $2$, namely $-I$, and that its elements outside $C_{ns}(\F_q)$ have order $4$. (Use
$(m_c\sigma)^2=N(c)I$.) For $q=3$ this recovers $Q_8$; what is the group for $q=5$?
\end{exercise}

\begin{exercise}\label{ex:groups:ex12}
(Harder.) Show that $N_s(\F_5)$ is a Sylow $2$-subgroup of $\GL_2(\F_5)$, and that its image in $\PGL_2(\F_5)\simeq S_5$, a dihedral group of order $8$, lies in a subgroup
isomorphic to $S_4$. (Show that $N_s(\F_5)$ and $\sm111{-1}$ generate a subgroup of order $96$ containing the scalars.)
\end{exercise}

\begin{exercise}\label{ex:groups:ex13}
Show that $I+3M_2(\Z_3)$ is torsion-free, and that $I+2M_2(\Z_2)$ contains elements of order $2$ (for instance $-I$ and $\mathrm{diag}(1,-1)$) but no element of order $4$.
(Hint: an element $g$ of order $4$ of $\GL_2(\Z_2)$ has eigenvalues $\pm\sqrt{-1}$, so its characteristic polynomial is $X^2+1$ and $g^2=-I$; if $g=I+2A$ and $g^2=-I$, then $4(A+A^2)=-2I$.)
\end{exercise}

\begin{exercise}\label{ex:groups:ex14}
Let $p\ne\ell$ be primes. Show that the image of a continuous homomorphism $\Z_p\to\GL_2(\F_\ell)$ is cyclic of $p$-power order, and deduce that it is trivial when
$p\nmid\#\GL_2(\F_\ell)$. (This is a first instance of the tame inertia arguments of \Cref{ch:local}.)
\end{exercise}
